\begin{frame}{The Graph Square Problem}
    \begin{problem}[SQUARE] \label{prob::square_problem}
        \leavevmode

        \textbf{Instance:} A finite, simple, undirected graph $G=(V,E)$.
        \medskip

        \textbf{Question:} Does there exist a finite, simple, undirected graph $H$ such that
        \[
        G = H^2?
        \]
    \end{problem}

    Let the \textbf{graph root} $H = (V, E_H)$ be a finite, simple, undirected graph, where $V$ is the vertex set with $|V| = n$, and $E_H \subseteq \{\{u,v\} \mid u,v \in V, u \neq v\}$ is the edge set.  
    The \textbf{square of a graph} $H$, denoted by $H^2 = (V, E_{H^2})$ is the graph on the same vertex set with edge set
    \begin{equation} \label{eq::fund_def_square}
      E_{H^2} = \{ uv \mid u \neq v, 1 \leq d_H(u,v) \leq 2 \}.
    \end{equation}
    When $G = H^2$ we say that $H$ is a \textbf{root} of $G$. 
\end{frame}

\begin{frame}{NP-Completeness}
    \citeauthor{motwani_sudan_1994_roots} (1994) proved that the graph square problem is $\mathbf{NP}$-complete by showing that the problem is $\mathbf{NP}$-hard through a polynomial reduction from Not-All-Equal $3$-SAT. 
    
    \begin{problem}[NAESAT] \label{prob::naesat_problem}
        \leavevmode

        \textbf{Instance:} A $3$-CNF formula
        \[
        \phi = C_1 \land C_2 \land \cdots \land C_m
        \]
        where each clause $C_i$ is over variables $x_1,\ldots,x_n$.

        \medskip

        \textbf{Question:} Does there exist an assignment of $\textsc{True}$ and $\textsc{False}$ to the variables such that every clause contains at least one $\textsc{True}$ literal and at least one $\textsc{False}$ literal?
    \end{problem}
\end{frame}

\begin{frame}{Maximal Cliques and Unicliqual Vertices in Tree Roots}
  \scriptsize
  \setlength{\itemsep}{0pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{0pt}
  \begin{algorithmic}[1]
    \State \textbf{Input:} A graph $G$ where $G = T^2$
    \State \textbf{Output:} A tree $T$

    \Statex \textbf{Step 1}
    \If{$G = K_p$ is complete}
      \State \textbf{Return} tree $T$ with depth $1$, root any vertex
    \Else
      \State \textbf{Continue}
    \EndIf

    \Statex \textbf{Step 2}
    \State Identify all maximal cliques $C_1, \dots, C_n$ of $G$ (e.g. Bron--Kerbosch \cite{bron_kerbosch_1973_maximal_cliques}) 

    \Statex \textbf{Step 3}
    \State Let $b_1, \dots, b_n$ be multicliqual points for $C_1, \dots, C_n$ 
    \State Build graph $T$ on $\{b_1,\dots,b_n\}$
    \For{each pair $(C_i, C_j)$ with $|C_i \cap C_j| = 2$}
      \State Add edge $(b_i, b_j)$ in $T$
    \EndFor
    \State  $T$ is a tree

    \Statex \textbf{Step 4}
    \For{each clique $C_i$}
      \State Let $n_i$ be the number of unicliqual points in $C_i$
      \State Attach $n_i$ leaves to $b_i$ in $T$
    \EndFor
    \State Resulting $T$ has $|V(G)|$ nodes and $T^2 = G$
  \end{algorithmic}
\end{frame}

\begin{frame}{Integer Program}
    Two distinct vertices are adjacent in $G=H^2$ if and only if they are connected by a path of length one or two in $H$.

    \begin{equation} \label{eq::fund_adj_square_or}
        A_G = (A_H)^2 \lor A_H.
    \end{equation}

     $\displaystyle (A_H)^2_{ij} = \big( (A_H)_{i1} \land (A_H)_{1j} \big) \lor \big( (A_H)_{i2} \land (A_H)_{2j} \big) \lor \cdots \lor \big( (A_H)_{in} \land (A_H)_{nj} \big)$

    \begin{equation} \label{eq::sat_square}
        g_{ij} \text{ if and only if } \big( \bigwedge_k (x_{ik} \land x_{kj}) \lor x_{ij} \big). 
    \end{equation}
    
    \begin{lemma} \label{lemma::G_equal_to_H^2}
        Let $G$ and $H$ be graphs with adjacency matrices $A_G$ and $A_H$,respectively. The graph $H$ is a root of $G$ if and only if 
        \[ 
            A_G = \mathbbm{1}_{\geq 1}\!\big((A_H)^2 - D_H + A_H\big).
        \]
    \end{lemma}
\end{frame}

\begin{frame}
    \scriptsize
    \begin{enumerate}[label=\roman*), itemsep=2pt, leftmargin=*]
        \item $x_{ij}\in\{0,1\}$ denotes the $(i,j)$ entry of the adjacency matrix $A_H$.
        \item $y_{ikj}\in\{0,1\}$ represents the product $x_{ik}x_{kj}$ in $(A_H)^2_{ij}$.
        \item $z_{ij}\in\mathbb{Z}_{\geq 0}$ represents the argument of the indicator function.
        \item $M \in \mathbb{Z}_{> 0}$ be a sufficiently large constant $M \leq n(n -1) = O(n^2)$.
    \end{enumerate}

    \vspace{0.1cm}

    \[
    D_H =
    \begin{bmatrix}
    \deg_H(1) & 0      & \cdots & 0 & 0      \\
    0         & \ddots & \vdots & \ddots & \vdots \\
    \vdots    & \cdots & \deg_H(i) & \cdots & 0      \\
    0         & \ddots & \vdots & \ddots & \vdots \\
    0         & \cdots & 0      & \cdots & \deg_H(n)
    \end{bmatrix}, \quad
    A_H =
    \begin{bmatrix}
    x_{11} & \cdots & x_{1j} & \cdots & x_{1n} \\
    \vdots  & \ddots  & \vdots & \ddots & \vdots  \\
    x_{i1} & \cdots & x_{ij} & \cdots & x_{in} \\
    \vdots  & \ddots  & \vdots & \ddots & \vdots  \\
    x_{n1} & \cdots & x_{nj} & \cdots & x_{nn}
    \end{bmatrix}
    \]
    \[
    (A_H)^2=
    \begin{bmatrix}
    \sum_k x_{1k}x_{k1}&\cdots&\sum_k x_{1k}x_{kj}&\cdots&\sum_k x_{1k}x_{kn}\\
    \vdots&\ddots&\vdots&\ddots&\vdots\\
    \sum_k x_{ik}x_{k1}&\cdots&\sum_k x_{ik}x_{kj}&\cdots&\sum_k x_{ik}x_{kn}\\
    \vdots&\ddots&\vdots&\ddots&\vdots\\
    \sum_k x_{nk}x_{k1}&\cdots&\sum_k x_{nk}x_{kj}&\cdots&\sum_k x_{nk}x_{kn}
    \end{bmatrix}.
    \]
\end{frame}

\begin{frame}
    \scriptsize
    \begin{equation}  \label{eq::fund_programming_equation}
        \begin{bmatrix}
        g_{11} \\
        \vdots \\
        g_{1j} \\
        \vdots \\
        g_{1n} \\
        \vdots \\
        g_{ij} \\
        \vdots \\
        g_{nn}
        \end{bmatrix} 
        =
        \begin{bmatrix}
        \mathbbm{1}_{\geq 1}\big(z_{11}\big) \\
        \vdots \\
        \mathbbm{1}_{\geq 1}\big(z_{1j}\big) \\
        \vdots \\
        \mathbbm{1}_{\geq 1}\big(z_{1n}\big) \\
        \vdots \\
        \mathbbm{1}_{\geq 1}\big(z_{ij}\big) \\
        \vdots \\
        \mathbbm{1}_{\geq 1}\big(z_{nn}\big)
        \end{bmatrix}
        =
        \begin{bmatrix} \displaystyle
        \mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{k1} - (D_H)_{11} + x_{11} \big) \\
        \vdots \\ \displaystyle
        \mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{kj} - (D_H)_{1j} + x_{1j} \big) \\
        \vdots \\ \displaystyle
        \mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{kn} - (D_H)_{1n} + x_{1n} \big)\\
        \vdots \\ \displaystyle
        \mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{ik}x_{kj} - (D_H)_{ij} + x_{ij} \big)\\
        \vdots \\ \displaystyle
        \mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{nk}x_{kn} - (D_H)_{nn} + x_{nn} \big)
        \end{bmatrix}.
    \end{equation}
\end{frame}

\begin{frame}
    \scriptsize
    \begin{alignat}{3}
        \text{Objective Function: } & \nonumber \\
        \max \quad & 0 \label{eq::integer_objective_function}
        \end{alignat}
        \begin{alignat}{3}
        \text{Variables: } & \nonumber \\ 
        & x_{ij} \in \mathbb{Z}_{\geq 0}, 
        && \forall i,j \in \{1, \dots, n\}, \label{eq::IP_x_variable} \\ 
        & y_{ikj} \in \mathbb{Z}_{\geq 0}, 
        && \forall i,k,j \in \{1, \dots, n\}, \label{eq::IP_y_variable} \\
        & z_{ij} \in \mathbb{Z}_{\geq 0}, 
        && \forall i,j \in \{1, \dots, n\}, \label{eq::IP_z_variable} \\
        \text{Constraints: } & \nonumber \\
        \text{(1) Indicator argument: } 
        & z_{ii} = \sum_{k = 1}^{n}y_{iki} - \sum_{k = 1}^{n} x_{ik}, 
        && \forall i \in \{1, \dots, n\}, \label{eq::IP_fund_matrix} \\
        & z_{ij} = \sum_{k = 1}^{n}y_{ikj} + x_{ij}, 
        && \forall i \neq j \in \{1, \dots, n\}, \label{eq::IP_indicator_arg} \\
        \text{(2) Indicator function: } 
        & z_{ij} \leq M g_{ij}, 
        && \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_upper} \\
        & z_{ij} \geq g_{ij}, 
        && \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_lower}\\
        \text{(3) Product: } 
        & y_{ikj} \leq x_{ik}, 
        && \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_first} \\
        & y_{ikj} \leq x_{kj}, 
        && \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_second} \\
        & y_{ikj} \geq x_{ik} + x_{kj} - 1, 
        && \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_lower} \\
        \text{(4) Upper Bound: } 
        & x_{ij} \leq 1, 
        && \forall i, j \in \{1, \dots, n\}, \label{eq::IP_x_upper_bound} \\
        & y_{ikj} \leq 1, 
        && \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_y_upper_bound} \\
        \text{(5) Symmetry: } 
        & x_{ij} = x_{ji}, 
        &&\quad \forall i,j \in \{1, \dots, n\}, \label{eq::IP_symmetry_x} \\
        \text{(6) Diagonal: } 
        & x_{ii} = 0, 
        &&\quad \forall i \in \{1, \dots, n\}.  \label{eq::IP_diagonal_x} 
    \end{alignat} \label{eq::integer_program}
\end{frame}

\begin{frame}{Methodology}
  \begin{itemize}
    \item Implemented in \textbf{SageMath}, with \textbf{Gurobi} as the optimization backend.
    \item Uses the \textbf{barrier method}, an interior-point method based on predictor--corrector Newton iterations.
    \item Graphs are considered up to isomorphism, but algorithms operate on labelled representatives generated by SageMath.
    \item Vertex labels are retained throughout each computation.
    \item Different labellings of the same graph may lead to different intermediate or selected optimization solutions.

\begin{frame}[fragile]{Linear Program}
    \begin{lstlisting}[
        style=python,
        basicstyle=\ttfamily\fontsize{4.5}{5}\selectfont,
        breaklines=true,
        aboveskip=0pt,
        belowskip=0pt,
        numbers=none,
        columns=fullflexible,
        keepspaces=true,
        caption={Linear Program.},
        label={lst::linear_program}
        ]
    def linear_program(G):
        m = MixedIntegerLinearProgram(solver='Gurobi', maximization=False)
        m.solver_parameter("Method", 2)
        x = m.new_variable(real=True, nonnegative=True)
        y = m.new_variable(real=True, nonnegative=True)
        z = m.new_variable(real=True, nonnegative=True)
        A_G = G.adjacency_matrix()
        n = A_G.nrows()
        M = n**2
        for i in range(n):
            for j in range(n):
                if i == j:
                    m.add_constraint(z[i,j] - (sum(y[i,k,j] for k in range(n)) - sum(x[i,k] for k in range(n)) + x[i,j]) == 0)
                else:
                    m.add_constraint(z[i,j] - (sum(y[i,k,j] for k in range(n)) + x[i,j]) == 0)
        for i in range(n):
            for j in range(n):
                m.add_constraint(z[i,j] - M * A_G[i][j] <= 0)
                m.add_constraint(-z[i,j] + A_G[i][j] <= 0)
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    m.add_constraint(y[i,k,j] - x[i,k] <= 0)
                    m.add_constraint(y[i,k,j] - x[k,j] <= 0)
                    m.add_constraint(- y[i,k,j] + x[i,k] + x[k,j] - 1 <= 0)
        for i in range(n):
            for j in range(n):
                m.add_constraint(x[i,j] == x[j,i])
        for i in range(n):
            m.add_constraint(x[i,i] == 0)
        for i in range(n):
            for j in range(n):
                m.add_constraint(x[i,j] - 1 <= 0)
                for k in range(n):
                    m.add_constraint(y[i,k,j] - 1 <= 0)
        return m, x
    \end{lstlisting}
\end{frame}

\begin{frame}
    The linear relaxation will never return an infeasible solution if $H$ exists. 

    \begin{proposition} \label{prop::linear_relaxation_feasibility}
        If $G$ has a square root $H$, then $\mathcal{LP}(G)$ is feasible. 
    \end{proposition}

    \begin{proof}
         Since $\mathcal{LP}(G)$ is obtained by relaxing the integrality constraints of $\mathcal{IP}(G)$, every feasible solution of $\mathcal{IP}(G)$ is also feasible for $\mathcal{LP}(G)$. Hence $\mathcal{LP}(G)$ is feasible.
    \end{proof}
\end{frame}

\begin{frame}{Fundamental Properties}
    \begin{lemma} \label{lemma::g_ab=0}
        Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b \in V_G$ such that $ab \notin E_G$. For every feasible solution $x$ of $\mathcal{LP}(G)$ and every $v \in V_G$ with $v \neq a,b$ we have $x_{ab} = y_{avb} = 0$. 
    \end{lemma}
    \begin{lemma} \label{lemma::g_ab=0,two_edge_sum}
        Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b \in V_G$ such that $ab \notin E_G$. For every feasible solution $x$ of $\mathcal{LP}(G)$ and every $v \in V_G$ with $v \neq a,b$ we have
        \[
        x_{av} + x_{vb} \leq 1.
        \]
    \end{lemma}
    \begin{lemma} \label{lemma::g_ab=0,x_av=1}
        Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b,v \in V_G$ such that $ab \notin E_G$. If a feasible solution $x$ of $\mathcal{LP}(G)$ satisfies $x_{av}=1$, then $x_{vb}=0$.
    \end{lemma}
\end{frame}

\begin{frame}{Sparse Pendants or Pendant Forks in Trees}
  \begin{lemma} \label{lemma::tree_pendant_pendant_fork}
    Every non-trivial tree with at least $3$ vertices contains either a \hyperref[def::sparse_pendant]{sparse pendant} or a \hyperref[def::pendant_fork]{pendant fork} subgraph.
  \end{lemma}

  \begin{proof}
    Root $T$ at any vertex. Let $s$ be a leaf of maximum depth, $u$ its parent. Every child of $u$ is a leaf, else a deeper vertex than $s$ would exist.
    \smallskip
    \textbf{Case 1: $u$ is not the root.} Let $v$ be the parent of $u$. Every neighbour of $u$ other than $v$ is a leaf child.
    \begin{itemize}
      \item If $u$ has exactly one leaf neighbour $s$, then $\deg_T(u)=2$, and $\{s,u,v\}$ induces a sparse pendant.
      \item If $u$ has two leaf neighbours $s_1,s_2$, then $\{s_1,s_2,u,v\}$ induces a pendant fork.
    \end{itemize}
    \smallskip
    \textbf{Case 2: $u$ is the root.} All neighbours of $u$ are leaf children, and $u$ has at least two of them.
    \begin{itemize}
      \item If $u$ has exactly two neighbours $s,v$, then $\{s,u,v\}$ induces a sparse pendant.
      \item Otherwise, pick distinct children $s_1,s_2,v$ of $u$; then $\{s_1,s_2,u,v\}$ induces a pendant fork. \qedhere
    \end{itemize}
  \end{proof}
\end{frame}

\begin{frame}{Leaf Fixing Linear Program}
    If $s$ is a leaf, $u$ is its corresponding stem, and there exists $v\in N_G[u]\setminus\{u\}$ such that
    \[
        N_G[s]\subset N_G[u]\subseteq N_G[v],
    \]
    with all vertices in $N_G[s]\setminus\{u,v\}$ having the same closed neighbourhood as $s$, then the corresponding root edges can be fixed by imposing
    \begin{equation} \label{eq::unicliqual_vertex_fix}
        x_{su}=x_{us}=x_{uv}=x_{vu}=1.
    \end{equation}

    \begin{itemize}
        \item \textbf{Sparse pendant:} $s$ belongs to a unique $K_3$ in $G$.
        \item \textbf{Pendant fork:} An induced $K_4=\{s_1,s_2,u,v\}$,
        where $s_1,s_2$ are leaves of $u$, satisfies
        \[
            N_G(s_1)\setminus K_4=N_G(s_2)\setminus K_4
            \subset N_G(u)\setminus K_4
            \subseteq N_G(v)\setminus K_4.
        \]
    \end{itemize}

    \begin{lemma} \label{lemma::unicliqual_vertex_fix}
        Let $G$ and $T$ be graphs such that $G = T^2$ and $T$ is a tree. $\max\mathcal{UVLP}(G)$ always fixes a sparse pendant or pendant fork subgraph of $T$.
    \end{lemma}
\end{frame}

\begin{frame}{Propagation of the Fixing}
    \begin{lemma}[$0$ Path Propagation] \label{lemma::0_path_propagation}
        Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,i,j,b \in V_G$ such that $a,i,j,b$ form a shortest induced path in $H$ between $a$ and $b$. For every feasible solution $x$ of $\mathcal{LP}(G)$,
        \[
        x_{ai}=1 \implies x_{ib}=0, \quad x_{ia}=1 \implies x_{bi}=0, \quad x_{jb}=1 \implies x_{aj}=0, \quad x_{bj}=1 \implies x_{ja}=0.
        \]
    \end{lemma}

    \begin{lemma}[$1$-Path Propagation] \label{lemma::1_path_propagation}
        Let $G$ be a graph and let $x$ be a feasible solution of $\mathcal{LP}(G)$. Suppose that $u-v-a_1$ form a shortest induced path in $H$ between $u$ and $a$ and that $x_{uv}=1$. Let $A=\{a_1,\ldots,a_r\}$ and $B=\{b_1,\ldots,b_s\}$ be such that every vertex $k \in V_G\setminus\{u,v\}$ satisfies
        $k \in A\cup B$, and suppose that 
        \[
        x_{um}=0 \text{ for every } m \in A, \quad g_{un}=0 \text{ for every } n \in B,
        \]
        and $g_{sa_1}=0$ for the vertex $s$ preceding $u$ on the shortest path to $a_1$. Then
        \[
        x_{va_1}=1.
        \]
    \end{lemma}
\end{frame}

\begin{frame}{Vertex Deletion}
    \scriptsize
    \centering
    \resizebox{\textwidth}{!}{
        \begin{minipage}{\textwidth}
        \begin{algorithmic}[1]
            \State \textbf{Input:} A graph $G$
            \State \textbf{Output:} A $LP$ model $(m,x)$
            \State Construct and solve $\mathcal{LP}(G)$.
            \If{$\mathcal{LP}(G)$ is integral}
                \State \Return $(m,x)$
            \EndIf
            \State Set $\displaystyle F=\left\{\{a,b\}\in\binom{V_G}{2}:x_{ab}=x_{ba}=1\right\}$.
            \State Compute $\displaystyle d_F(u)=\left|\left\{w\in N_G(u):\{u,w\}\notin F\right\}\right|$ for each $u\in V_G$.
            \State Choose $v\in\arg\min_{u\in V_G}d_F(u)$ and set $V^-=V_G \setminus\{v\}$.
            \State Construct $G^-=G[V^-]$ and inherit the fixing $\displaystyle F^-=\left\{\{a,b\}\in F:a,b\in V^-\right\}$.
            \State Construct and solve $\mathcal{LP}(G^-)+F^-$.
            \If{$\mathcal{LP}(G^-)+F^-$ is integral}
                \State Construct $H^-$ from its integral solution.
            \Else
                \State \Return $(m,x)$
            \EndIf
            \State Construct a new copy of $\mathcal{LP}(G)$ and impose $x_{ab}=x_{ba}=1$ for every $\{a,b\}\in F$ and every $ab\in E_{H^-}$.
            \State Solve the modified LP.
            \If{the modified $LP$ is infeasible}
                \State \Return $(m,x)$
            \Else
                \State \Return the modified $LP$ solution $(m,x)$
            \EndIf
        \end{algorithmic}
        \end{minipage}
    }
\end{frame}

\begin{frame}{Symbolic Sets}
    \scriptsize
    \begin{conjecture} \label{conjecture::two_edges_linear_maximization}
        Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b \in V_G$ such that $ab \notin E_G$. For every feasible solution $x$ of $\max \mathcal{LP}(G)$ and every $v \in V_G$ with $v \neq a,b$ we have
        \[
        x_{av} + x_{vb} = 1 \text{ or }  x_{av} + x_{vb} = 0. 
        \]
    \end{conjecture}
    \begin{definition}[Consistent Set] \label{def::consistent_set}
        Let $E'\subseteq E_G$ be a maximal subset with $|E'|\geq 3$ such that, for every $e\in E'$, there exists $f \in E'$ such that $e$ and $f$ satisfy the $=1$ relationship. We call $E'$ a \textbf{consistent set} if there exists a partition
        \[
        E'=A\cup B,\quad A\cap B=\varnothing,
        \]
        into nonempty sets $A$ and $B$ such that, for all $e,e'\in A$, the edges $e$ and $e'$ do not form a $P_3$ in $G$, and for all $f,f'\in B$, the edges $f$ and $f'$ do not form a $P_3$ in $G$.
    \end{definition}
    \begin{definition}[Inconsistent Set] \label{def::inconsistent_set}
        Let $E'\subseteq E_G$ be a maximal subset with $|E'|\geq 3$ such that, for every $e\in E'$, there exists $f\in E'$ such that $e$ and $f$ satisfy the $=1$ relationship. We call $E'$ an \textbf{inconsistent set} if there is no partition
        \[
        E'=A\cup B,\quad A\cap B=\varnothing,
        \]
        into nonempty sets $A$ and $B$ such that, for all $e,e'\in A$, the edges $e$ and $e'$ do not form a $P_3$ in $G$, and, for all $f,f'\in B$, the edges $f$ and $f'$ do not form a $P_3$ in $G$.

\begin{frame}{Rounding}
    \scriptsize
    \centering
    \resizebox{\textwidth}{!}{
        \begin{minipage}{\textwidth}
            \begin{algorithmic}[1]
                \State \textbf{Input:} A graph $G$
                \State \textbf{Output:} A $LP$ model $(m,x)$

                \vspace{0.5em}

                \State Initialize $F=\emptyset$ and construct $\mathcal{UVLP}(G)$.
                \State Solve $\mathcal{UVLP}(G)$.
                \While{the current $LP$ solution is fractional}
                \State For each fractional edge $ab$, compute its rounding score $ s_{ab}=|x_{ab}-0.5|$.
                \State Select a fractional edge $ab$ with maximum score.
                \State Set
                \[
                \widehat{x}_{ab}=\widehat{x}_{ba}=
                \begin{cases}
                0, & x_{ab} < 0.5,\\
                1, & x_{ab} \geq 0.5.
                \end{cases}
                \]
                \State Add $ab$ with its assigned value to $F$.
                \State Reconstruct $\mathcal{UVLP}(G)$ with all fixings in $F$.
                \State Solve $\mathcal{UVLP}(G) + F$.
                \If{$\mathcal{UVLP}(G) + F$ is infeasible}
                    \State \textbf{return} the previous feasible solution $(m,x)$.
                \EndIf
                \If{$\mathcal{UVLP}(G) + F$ is integral}
                    \State \textbf{return} the integral solution $(m,x)$.
                \EndIf
                \EndWhile
                \State \textbf{return} $(m,x)$.
            \end{algorithmic}
