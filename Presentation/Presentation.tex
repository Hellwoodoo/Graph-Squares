\begin{frame}{The Graph Square Problem}
    \begin{problem}[SQUARE] \label{prob::square_problem}
        \leavevmode

        \textbf{Instance:} A finite, simple, undirected graph $G=(V,E)$.
        \medskip

        \textbf{Question:} Does there exist a finite, simple, undirected graph $H$ such that
        \[
        G = H^2?
        \]
    \end{problem}

    Let the \textbf{graph root} $H = (V, E_H)$ be a finite, simple, undirected graph, where $V$ is the vertex set with $|V| = n$, and $E_H \subseteq \{\{u,v\} \mid u,v \in V, u \neq v\}$ is the edge set.  
    The \textbf{square of a graph} $H$, denoted by $H^2 = (V, E_{H^2})$ is the graph on the same vertex set with edge set
    \begin{equation} \label{eq::fund_def_square}
      E_{H^2} = \{ uv \mid u \neq v, 1 \leq d_H(u,v) \leq 2 \}.
    \end{equation}
    That is, two distinct vertices are adjacent in $H^2$ if and only if they are connected by a path of length one or two in $H$. When $G = H^2$ we say that $H$ is a \textbf{root} of $G$. 
\end{frame}

\begin{frame}{NP-Completeness}
    \citeauthor{motwani_sudan_1994_roots} (1994) proved that the graph square problem is $\mathbf{NP}$-complete by showing that the problem is $\mathbf{NP}$-hard through a polynomial reduction from Not-All-Equal $3$-SAT. 
    
    \begin{problem}[NAESAT] \label{prob::naesat_problem}
        \leavevmode

        \textbf{Instance:} A $3$-CNF formula
        \[
        \phi = C_1 \land C_2 \land \cdots \land C_m
        \]
        where each clause $C_i$ is over variables $x_1,\ldots,x_n$.

        \medskip

        \textbf{Question:} Does there exist an assignment of $\textsc{True}$ and $\textsc{False}$ to the variables such that every clause contains at least one $\textsc{True}$ literal and at least one $\textsc{False}$ literal?
    \end{problem}
\end{frame}

\begin{frame}{Maximal Cliques and Unicliqual Vertices in Tree Roots}
  \scriptsize
  \setlength{\itemsep}{0pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{0pt}
  \begin{algorithmic}[1]
    \State \textbf{Input:} A graph $G$ where $G = T^2$
    \State \textbf{Output:} A tree $T$

    \Statex \textbf{Step 1}
    \If{$G = K_p$ is complete}
      \State \textbf{Return} tree $T$ with depth $1$, root any vertex
    \Else
      \State \textbf{Continue}
    \EndIf

    \Statex \textbf{Step 2}
    \State Identify all maximal cliques $C_1, \dots, C_n$ of $G$ (e.g. Bron--Kerbosch \cite{bron_kerbosch_1973_maximal_cliques}) 

    \Statex \textbf{Step 3}
    \State Let $b_1, \dots, b_n$ be multicliqual points for $C_1, \dots, C_n$ 
    \State Build graph $T$ on $\{b_1,\dots,b_n\}$
    \For{each pair $(C_i, C_j)$ with $|C_i \cap C_j| = 2$}
      \State Add edge $(b_i, b_j)$ in $T$
    \EndFor
    \State  $T$ is a tree

    \Statex \textbf{Step 4}
    \For{each clique $C_i$}
      \State Let $n_i$ be the number of unicliqual points in $C_i$
      \State Attach $n_i$ leaves to $b_i$ in $T$
    \EndFor
    \State Resulting $T$ has $|V(G)|$ nodes and $T^2 = G$
  \end{algorithmic}
\end{frame}

\begin{frame}{Integer Program}
    \begin{equation} \label{eq::fund_adj_square_or}
        A_G = (A_H)^2 \lor A_H.
    \end{equation}

     $\displaystyle (A_H)^2_{ij} = \big( (A_H)_{i1} \land (A_H)_{1j} \big) \lor \big( (A_H)_{i2} \land (A_H)_{2j} \big) \lor \cdots \lor \big( (A_H)_{in} \land (A_H)_{nj} \big)$

    \begin{equation} \label{eq::sat_square}
        g_{ij} \text{ if and only if } \big( \bigwedge_k (x_{ik} \land x_{kj}) \lor x_{ij} \big). 
    \end{equation}
    
    \begin{lemma} \label{lemma::G_equal_to_H^2}
        Let $G$ and $H$ be graphs with adjacency matrices $A_G$ and $A_H$,respectively. The graph $H$ is a root of $G$ if and only if 
        \[ 
            A_G = \mathbbm{1}_{\geq 1}\!\big((A_H)^2 - D_H + A_H\big).
        \]
    \end{lemma}
\end{frame}
