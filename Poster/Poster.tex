\headerbox{Some Instructions}{name=Subtitle4,column=0,span = 2, below=Subtitle1}{ 
\vspace{0.5em}
\begin{enumerate}\compresslist
    \item Structure your poster into sections, including an introduction
    \vspace{0.25em}
    %\item Give some context and chose two or three key references
    \vspace{0.25em}
    \item Include figures and graphics; make sure they have captions 
    \vspace{0.25em}
    \item Highlight your main findings
    \vspace{0.25em}
    \item Technical abbreviations can be used but should be written out once.
    \vspace{0.25em}
    \item Try using bullets points rather than long text: {\em less can be more.}
    \vspace{0.25em}
    \item This template serves as an example: {\em You can modify it as you like.}
     \vspace{0.25em}
    \item Distill important information in figures or tables with \LaTeX; e.g.
\begin{center}
\begin{tabular}{l c c}
\toprule
\textbf{Group} & \textbf{Response 1} & \textbf{Response 2}\\
\midrule
Group 1 & 1.1 & 0.56 \\
Group 2 & 3 & 0.9 \\
Group 3 & 2 & 0.2 \\
\bottomrule
\end{tabular}
\captionof{table}{Table caption}
\end{center}
%\item Use a constant font throughout the poster.    
\end{enumerate}
