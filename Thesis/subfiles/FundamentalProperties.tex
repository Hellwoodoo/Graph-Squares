\begin{proposition} \label{prop::g_ab=0}
  Consider the graph square $G$ where the edge between vertices $a$ and $b$ is missing, $ab \notin E_G$ such that $g_{ab} = 0$. Then every feasible solution of $\mathcal{LP}(G)$ satisfies $x_{ab} = 0$ and $y_{avb} = 0$ for all vertices $v \in V \backslash \{a,b\}$. 
\end{proposition}
\begin{proof}
  Using Equations~\eqref{eq::IP_indicator_upper}--\eqref{eq::IP_indicator_lower} if $g_{ab} = 0$ then $z_{ab} \leq 0$ and $z_{ab} \geq = 0$ such that $z_{ab} = 0$. Then Equation~\eqref{eq::IP_indicator_arg} gives that $\displaystyle 0 = \sum_{v = 1}^{n}y_{avb} + x_{ab}$. Since each $x,y$ variable are non-negative, this only holds when each variable is equal to $0$ which gives that $x_{ab} = 0$ and $y_{avb} = 0$ for all vertices $v \in V \backslash \{a,b\}$. 
\end{proof}

\begin{proposition} \label{prop::g_ab=0,two_edge_sum}
  Consider the graph square $G$ where the edge between vertices $a$ and $b$ is missing, $ab \notin E_G$ such that $g_{ab} = 0$. Then every feasible solution of $\mathcal{LP}(G)$ satisfies that for all vertices $v \in V \backslash \{a,b\}$, $x_{av} + x_{vb} \leq 1$. 
\end{proposition}
\begin{proof}
  Since Proposition~\eqref{prop::g_ab=0} gives $y_{avb}=0$, using Equation~\eqref{eq::IP_indicator_lower} gives that $0 = y_{avb} \geq x_{av} + x_{vb} - 1$ which after rearranging gives $x_{av} + x_{vb} \leq 1$ as required. 
\end{proof}
