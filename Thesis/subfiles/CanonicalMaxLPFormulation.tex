\begin{alignat}{3}
\text{Objective Function: } & \nonumber \\
& \min - \sum_{i=1}^{n} \sum_{j=1}^{n} x_{ij}. \label{eq::primal_max_obj_function}
\end{alignat}
\begin{alignat}{3}
\text{Variables: } & \nonumber \\ 
& x_{ij} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::canonical_LP_x_variable} \\
& y_{ikj} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,k,j \in \{1, \dots, n\}, \label{eq::canonical_LP_y_variable} \\
& z_{ij} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\} \label{eq::canonical_LP_z_variable}. \\
\text{Constraints: } & \nonumber \\
\text{(1) Indicator argument: } 
& z_{ii} - \sum_{k = 1}^{n}y_{iki} + \sum_{k = 1}^{n} x_{ik} - x_{ii} = 0, 
&& \forall i \in \{1, \dots, n\} , \label{eq::canonical_LP_fund_matrix} \\
& z_{ij} - \sum_{k = 1}^{n}y_{ikj} - x_{ij} = 0, 
&& \forall i \neq j \in \{1, \dots, n\} , \label{eq::canonical_LP_indicator_arg} \\
\text{(2) Indicator function: } 
& z_{ij} - M g_{ij} \leq 0, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::canonical_LP_indicator_upper} \\
& - z_{ij} + g_{ij} \leq 0, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::canonical_LP_indicator_lower}\\
\text{(3) Product: } 
& y_{ikj} - x_{ik} \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::canonical_LP_product_upper_first} \\
& y_{ikj} - x_{kj} \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::canonical_LP_product_upper_second} \\
&  x_{ik} + x_{kj} - y_{ikj} - 1 \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::canonical_LP_product_lower} \\
\text{(4) Symmetry: } 
& x_{ij} - x_{ji} = 0, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::canonical_LP_symmetry_x} \\
\text{(5) Diagonal: } 
& x_{ii} = 0, 
&&\qquad \forall i \in \{1, \dots, n\}, \label{eq::canonical_LP_diagonal_x} \\
\text{(6) Variable Bound: } 
& x_{ij} - 1 \leq 0,
&& \forall i, j \in \{1, \dots, n\}, \label{eq::canonical_LP_x_upper_bound} \\
& y_{ikj} - 1 \leq 0,
&& \forall i,k, j \in \{1, \dots, n\} \label{eq::canonical_LP_y_upper_bound}.
\end{alignat} \label{eq::canonical_linear_program}
