\begin{lemma}[Propagation] \label{lemma::propagation}
  Let $G$ and $T$ be graphs such that $G=T^2$ and $T$ is a tree. Let $s,u,v \in V_G$ be identified such that $s$ is a leaf, $u$ is its stem, and $v$ is the external vertex, with
  \[
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
  \]
  Let $N_T(v)\setminus{u} = \{a_1,\ldots,a_r\}$ be the set of all other neighbours of $v$ in $T$. Then path propagation fixes
  \[
  x_{va_i}=x_{a_iv}=1 \text{ for every } i \in \{1,\ldots\}. 
  \]
\end{lemma}
\begin{proof}
  Since $s$ is a leaf and $u$ is a stem of $T$, the path $s-u-v$ is contained in $T$. Let $A=\{a_1,\ldots,a_r\}$ denote the vertices adjacent to $v$ in $T$ other than $u$, and let $B=\{b_1,\ldots,b_s\}$ denote the vertices adjacent to a vertex in $A$ other than $v$. For each $a_i\in A$, the vertices $s,u,v,a_i$ form a $P_4$ in $T$. Since $x_{su}=1$ and there exists no cycles in $T$, Lemma~\eqref{lemma::0_path_propagation} gives
  \[
  x_{ua_i}=0 \quad \text{ for all } a_i \in A.
  \]
  Without loss of generality consider $a_1\in A$. Since $ua_1\in E_G$, have $g_{ua_1}=1$, and hence Equation~\eqref{eq::IP_indicator_lower} gives $z_{ua_1}\geq1$. Since $x_{ua_1}=0$, Equation~\eqref{eq::IP_indicator_arg} gives
  \[
  1 \leq z_{ua_1} = x_{ua_1}+\sum_{k\in V_G}y_{uka_1} = \sum_{k\in V_G}y_{uka_1}.
  \]
  The terms corresponding to $k=s$ and $k=u$ are zero since $g_{sa_1}=g_{uu}=0$, and hence Lemma~\eqref{lemma::g_ab=0} gives $x_{sa_1}=x_{uu}=0$. Therefore, by Equations~\eqref{eq::IP_product_upper_first}--\eqref{eq::IP_product_lower},
  \[
  y_{usa_1}=y_{uua_1}=0.
  \]
  For every $m\in A$, we have $x_{um}=0$, and hence Equation~\eqref{eq::IP_product_upper_first} gives
  \[
  y_{uma_1}=0.
  \]
  Similarly, for every $n\in B$, we have $g_{un}=0$, and therefore Lemma~\eqref{lemma::g_ab=0} gives $x_{un}=0$. Thus Equation~\eqref{eq::IP_product_upper_first} gives
  \[
  y_{una_1}=0.
  \]
  Thus all terms in the sum are zero except possibly $y_{uva_1}$, giving
  \[
  1 \leq y_{uva_1}.
  \]
  Since $x_{uv}=1$, Equation~\eqref{eq::IP_product_upper_first} gives
  $y_{uva_1}\leq x_{uv}=1$, while Equation~\eqref{eq::IP_product_upper_second} gives $y_{uva_1} \leq x_{va_1}$. Hence
  $1 \leq y_{uva_1} \leq x_{va_1} \leq 1$, and therefore
  \[
  x_{va_1}=1.
  \]
  This process can be repeated for each $a_i\in A$, giving $x_{va_i}=1$ for all $i\in\{1,\ldots,r\}$. Thus the edges $uv, va_1,\ldots,va_r$ are identified in the root, with $u$ acting as a leaf, $v$ as its stem, and each $a_i$ as the next vertex along the tree.
\end{proof}

\begin{theorem}[Inductive Proof] \label{theorem::unicliqual_vertex_induction}
  Let $G$ and $T$ be graphs such that $G=T^2$ and $T$ is a tree. Then $\max \mathcal{UVLP}(G)$ returns an integral solution.
\end{theorem}

\begin{proof}

  Suppose that $G=T^2$ for some tree $T$. By Lemma~\ref{lemma::unicliqual_vertex_fix}, the unicliqual vertex fixing
  identifies a leaf $s$, its stem $u$, and the external vertex $v$, and Equation~\eqref{eq::unicliqual_vertex_fix} imposes
  \[
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
  \]
  Hence the hypotheses of Lemma~\ref{lemma::propagation} are satisfied.

  We prove by induction on the number of edges of the reconstructed subtree that repeated application of path propagation assigns an integral value to every edge of $T$.

  For the base case, the reconstructed subtree consists of the edges initially identified by the unicliqual vertex fixing. These edges have integral assignment by Equation~\eqref{eq::unicliqual_vertex_fix}.

  For the inductive step, suppose that a connected subtree $T'$ of $T$ containing the fixed path $s-u-v$ has been reconstructed and that every edge of $T'$ has been assigned its integral value. If $T'\neq T$, then, since $T$ is connected, there exists an edge
  \[
  ab \in E_T\setminus E_{T'}
  \]
  such that $a\in V_{T'}$ and $b\notin V_{T'}$.

  Since $T$ is a tree, the edge $ab$ extends $T'$ without creating a cycle. Moreover, the unique path in $T$ from the fixed path $s-u-v$ to $b$ consists of edges already reconstructed up to $a$, followed by the edge $ab$. Thus path propagation can be applied to this extended path, and fixes
  \[
  x_{ab}=x_{ba}=1.
  \]
  Consequently, $T'$ can be extended to the larger connected subtree $T''=T'+ab$, whose edges all have integral assignments.

  Therefore, whenever the reconstructed subtree is a proper subtree of $T$, path propagation identifies at least one additional edge of $T$. Since $T$ is finite, repeated application of this argument eventually reconstructs all of $T$ and assigns an integral value to every edge.

  Hence the resulting solution is integral and corresponds to the tree root $T$ of $G$. Therefore $\max\mathcal{UVLP}(G)$ returns an integral solution.
\end{proof}

\begin{theorem}[Path Proof] \label{theorem::unicliqual_vertex_path}
  Let $G$ and $T$ be graphs such that $G=T^2$ and $T$ is a tree. Then $\max \mathcal{UVLP}(G)$ returns an integral solution.
\end{theorem}
\begin{proof}
  Suppose that $G=T^2$ for some tree $T$. By Lemma~\ref{lemma::unicliqual_vertex_fix}, the unicliqual vertex fixing identifies a leaf $s$, its stem $u$, and the external vertex $v$, and Equation~\eqref{eq::unicliqual_vertex_fix} imposes
  \[
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
  \]
  Hence the hypotheses of Lemma~\ref{lemma::propagation} are satisfied.

  In particular letting $N_T(v)\setminus\{u\}=\{a_1,\ldots,a_r\}$ Lemma~\ref{lemma::propagation} fixes
  \[
  x_{va_i}=x_{a_iv}=1 \text{ for every }i\in\{1,\ldots,r\}.
  \]
  The same argument can then be applied at each newly identified vertex. More generally, let $w\in V(T)$ be arbitrary. Since $T$ is connected, there is a path from $v$ to $w$. Therefore, together with the fixed path $s-u-v$, there is a path
  \[
  s-u-v-\cdots-w
  \]
  in $T$. Starting from the fixed edges $su$ and $uv$, path propagation can be applied successively along this path. Hence every edge on the path from $v$ to $w$ is assigned an integral value.

  Since $w$ was arbitrary, every vertex of $T$ is reached by such a path. Now let $ab\in E_T$ be arbitrary. Since $T$ is connected, there is a path from $v$ to $a$, and appending the edge $ab$ gives a path
  \[
  s-u-v-\cdots-a-b
  \]
  containing $ab$. Thus path propagation assigns an integral value to $ab$.

  Therefore every edge of $T$ is assigned an integral value. Hence the resulting assignment is integral and corresponds to the tree root $T$ of $G$. Therefore $\max\mathcal{UVLP}(G)$ returns an integral solution.
\end{proof}
