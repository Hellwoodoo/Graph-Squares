\begin{abstract}
In this paper, we investigate the matrix properties of graph squares, focusing in particular on the
relationship between the adjacency matrix of a graph and the corresponding adjacency of its square. 
\end{abstract}

\tableofcontents

\section{Introduction} \label{sec::intro}

The problem of finding the square root of a graph is a well-studied field in both mathematics and computer-science, 
with specific relation to distributed computing, network design and computational biology. Research in this area has
primarily focused around the undirected graph $H$ and its square $G = H^2$ which is obtained by adding an edge between 
every pair of vertices whose distance in $H$ is at most two. 

Much of the existing research on graph square roots has centered on characterizing when a graph admits such 
a root and on the complexity of reconstructing $H$ from $G$. In contrast, rather than studying the construction problem 
directly, we analyze the relationship between the adjacency matrix $A_H$ of the root and the adjacency matrix $A_G$ of 
the square, and show that the problem can be completely categorized as an integer program. 

We then examine the relaxed linear program which can be evaluated in polynomial time using interior point method, and 
integrate existing literature on special root structures as a stepping-stone towards generating a true polynomial time 
algorithm. 

\section{Background} \label{sec::background}
Let the \emph{graph root} $H = (V, E_H)$ be a finite, simple, undirected graph, where $V$ is the vertex set with $|V| = n$, 
and $E_H \subseteq \{\{u,v\} \mid u,v \in V,\; u \neq v\}$ is the edge set. Denote by $d_H(u,v)$ the 
\emph{graph distance} defined as the number of edges of the shortest path 
between vertices $u$ and $v$ in $H$. If no such path exists, $d_H(u,v) = \infty$.

The \emph{square} of the graph $H$, denoted by $H^2 = (V, E_{H^2})$ is the graph on the same vertex set with edge set
\begin{equation} \label{eq::fund_def_square}
E_{H^2} = \{\, uv \mid u \neq v,\; 1 \leq d_H(u,v) \leq 2 \,\}.
\end{equation}
That is, two distinct vertices are adjacent in $H^2$ if and only if they are connected by a path of length one or two in $H$. 
This is the standard definition of the square of a graph.

Using an alternative construction, let the \emph{graph square} $G = (V, E_G)$ be an undirected graph defined on 
the same vertex set $V$. Let $A_H \in \{0,1\}^{n \times n}$ denote the adjacency matrix of $H$, where
\[
(A_H)_{ij} =
\begin{cases}
1 ,& \text{if } ij \in E_H, \\
0 ,& \text{otherwise},
\end{cases}
\]
and let $D_H \in \mathbb{R}^{n \times n}$ denote the degree matrix of $H$, where
\[
(D_H)_{ij} =
\begin{cases}
\deg_H(i) ,& \text{if } i = j, \\
0 ,& \text{otherwise}.
\end{cases}
\]
Let $\mathbbm{1}_{\geq 1} : \mathbb{R}^{n \times n} \to \{0,1\}^{n \times n}$ denote the 
entrywise \emph{indicator function}, defined by
\[
\mathbbm{1}_{\geq 1}(M)_{ij} =
\begin{cases}
1 ,& \text{if } M_{ij} \geq 1, \\
0 ,& \text{otherwise}.
\end{cases}
\]
Define the adjacency matrix of $G$ using the adjacency and degree matrices of $H$
\begin{equation} \label{eq::fund_adj_eq_undirected}
A_G = \mathbbm{1}_{\geq 1}\!\big((A_H)^2 - D_H + A_H\big).
\end{equation}
\begin{lemma} \label{lemma::G_equal_to_H^2}
The graph $G$ defined by \eqref{eq::fund_adj_eq_undirected} is equivalent to the graph square $H^2$
 defined in \eqref{eq::fund_def_square} 
\begin{equation} \label{eq::fund_def_square_expanded}
 ij \in E_{H^2} \iff 1 \leq d_H(i,j) \leq 2. 
\end{equation}
\end{lemma}
\begin{proof} 
\noindent
For $i \neq j$, since $D_H$ is diagonal, have $(D_H)_{ij} = 0$ and hence
\[
\big((A_H)^2 - D_H + A_H\big)_{ij} = (A_H)^2_{ij} + (A_H)_{ij}.
\]

($\Rightarrow$) Suppose $ij \in E_G$ such that $(A_H)^2_{ij} + (A_H)_{ij} \geq 1$.
This implies that at least on of $(A_H)_{ij} = 1$ or $(A_H)^2_{ij} \geq 1$ must be true. If $(A_H)_{ij} = 1$, then $d_H(i,j) = 1$.  
If $(A_H)^2_{ij} \geq 1$, then there exists a vertex $k$ such that $ik, kj \in E_H$, and hence $d_H(i,j) \leq 2$. Therefore 
$1 \leq d_H(i,j) \leq 2$, and thus $ij \in E_{H^2}$.

($\Leftarrow$) Suppose $ij \in E_{H^2}$ such that $1 \leq d_H(i,j) \leq 2$. 
$D_H$ is diagonal such that $(D_H)_{ij} = 0$. If $d_H(i,j) = 1$, then $(A_H)_{ij} = 1$.  
If $d_H(i,j) = 2$, then there exists a vertex $k$ such that $ik, kj \in E_H$, and hence $(A_H)^2_{ij} \geq 1$.
In either case, $(A_H)^2_{ij} + (A_H)_{ij} \geq 1$, so $(A_G)_{ij} = 1$, and hence $ij \in E_G$.

For $i = j$, since all graphs are simple, have $(E_{\cdot})_{ii} = 0$, $(A_{\cdot})_{ii} = 0$ and hence
\[
\big((A_H)^2 - D_H + A_H\big)_{ii} = (A_H)^2_{ii} - (D_H)_{ii}.
\]
$(A_H)^2_{ii} - (D_H)_{ii} = 0$ is always true since the degree matrix of $H$ always has diagonal 
values $\displaystyle (D_H)_{ii} = \sum_k x_{ik}$ and each edge from vertex $i$ to its neighbour in $H$ 
will always create a loop to itself in the square such that $\displaystyle (A_H)^2_{ii} = 
\sum_{k = 1} x_{ik}x_{ki} = \sum_{k = 1} x_{ik}^2$ by the symmetry of adjacency matrix $A_H$. Since $x_{ik} \in \{0, 1\}$,
the square of the boolean is identical to the boolean $x_ik$ such that $\displaystyle (A_H)^2_{ii} = \sum_k x_{ik} = (D_H)_{ii}$
and $(A_G)_{ii} = \mathbbm{1}_{\geq 1}\!\big((A_H)^2 - D_H + A_H\big)$ as required. 

Combining both gives that $E_G = E_{H^2}$, and so $G = H^2$.  

\end{proof}
\noindent
The formulation of the adjacency matrix of the graph square in Equation~\eqref{eq::fund_adj_eq_undirected} is very powerful. 
However, the presence of the indicator function introduces a nonlinearity that prevents the problem from being expressed 
as a purely quadratic form, thereby limiting the direct applicability of existing results from the quadratic optimization literature.

\begin{lemma} \label{lemma::no_indicator_girth_5}
  If there are no $K_3$ triangle subgraphs or any $C_4$ square subgraphs in the graph root such that $H$ has girth of length $5$,
  then  Equation~\eqref{eq::fund_adj_eq_undirected} can be simplified to become 
  \begin{equation} \label{eq::fund_adj_eq_undirected_no_indicator}
    A_G = \big((A_H)^2 - D_H + A_H\big).
  \end{equation}
\end{lemma}

\begin{proof}
  Since the graph root has girth of length $5$ there are no cycles of length $3$ or $4$ in $H$. 
  Thus for every pair of vertices $i \neq j \in V$ there is either no cycle that connects $i,j$ or there
  is a cycle of at least length $5$ between them. In all cases $(D_H)_{ij} = 0$. 
  If there is no cycle, then there is at most one path between $i,j$
  \begin{enumerate}[label=\roman*)]
    \item If the path is of length $1$, then $(A_H)_{ij} = 1, (A_H)^2_{ij} = 0 \implies (A_G)_{ij} = 0 - 0 + 1 = 1$.
    \item If the path is of length $2$, then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 1 \implies (A_G)_{ij} = 1 - 0 - 0 = 1$.
    \item If the path is of length $\geq 3$ or a path does not exist, then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 0 \implies (A_G)_{ij} = 0$.
  \end{enumerate}
  If there is a cycle of length $\geq 5$ between $i,j$ such that there are two paths between $i,j$
  \begin{enumerate}[label=\roman*)]
    \item If one path is of length $1$, then the other is of length $\geq 4$ and hence $(A_H)_{ij} = 1, (A_H)^2_{ij} = 0 \implies (A_G)_{ij} = 0 - 0 + 1 = 1$.
    \item If one path is of length $2$, then the other is of length $\geq 3$ and hence $(A_H)_{ij} = 0, (A_H)^2_{ij} = 1 \implies (A_G)_{ij} = 1 - 0 - 0 = 1$.
    \item If the paths are both of length $\geq 3$,  then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 0 \implies (A_G)_{ij} = 0$. 
  \end{enumerate}
  If $i = j$, then \((A_H)_{ii} = 0\), \((D_H)_{ii} = \deg_H(i)\), and \((A_H)^2_{ii} = \deg_H(i)\). Thus,
  \[
  (A_H)^2_{ii} - (D_H)_{ii} + (A_H)_{ii}
  = \deg_H(i) - \deg_H(i) + 0 = 0 = (A_G)_{ii}.
  \]
\end{proof}

\subsection{Integer Program} \label{subsec::IP}
Let $x_{ij} \in \{0, 1\}$ be the decision variables of $A_H$, let $y_{ikj} \in \{0, 1\}$ be the decision variable for 
$x_{ik}x_{kj}$ in $(A_H)^2_{ij}$ and let $z_{ij} \in \mathbb{Z}_{\geq 0}$ represent the argument inside the indicator function. 

$\displaystyle 
D_H =
\begin{bmatrix}
\deg_H(1) & 0      & \cdots & 0 & 0      \\
0         & \ddots & \vdots & \ddots & \vdots \\
\vdots    & \cdots & \deg_H(i) & \cdots & 0      \\
0         & \ddots & \vdots & \ddots & \vdots \\
0         & \cdots & 0      & \cdots & \deg_H(n)
\end{bmatrix}, 
$ 
$\displaystyle A_H =
\begin{bmatrix}
x_{11} & \cdots & x_{1j} & \cdots & x_{1n} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
x_{i1} & \cdots & x_{ij} & \cdots & x_{in} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
x_{n1} & \cdots & x_{nj} & \cdots & x_{nn}
\end{bmatrix}
$ such that 
\[
(A_H)^2 = \begin{bmatrix}
\displaystyle \sum_{k=1}^n x_{1k}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{1k}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{1k}x_{kn} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
\displaystyle \sum_{k=1}^n x_{ik}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{ik}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{ik}x_{kn} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
\displaystyle \sum_{k=1}^n x_{nk}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{nk}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{nk}x_{kn}
\end{bmatrix}.
\]
Then with known constants $g_{ij} \in \{0, 1\}$ from $A_G$, can reformulating \eqref{eq::fund_adj_eq_undirected} as an 
integer program by equating entries of each matrix to give the following linear system of $n^2$ equations.
\begin{equation}  \label{eq::fund_programming_equation}
\begin{bmatrix}
g_{11} \\
\vdots \\
g_{1j} \\
\vdots \\
g_{1n} \\
\vdots \\
g_{ij} \\
\vdots \\
g_{nn}
\end{bmatrix} 
=
\begin{bmatrix}
\mathbbm{1}_{\geq 1}\big(z_{11}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{1j}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{1n}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{ij}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{nn}\big)
\end{bmatrix}
=
\begin{bmatrix} \displaystyle
\mathbbm{1}_{\geq 1}\!\big(\sum_{k=1}^n x_{1k}x_{k1} - (D_H)_{11} + x_{11} \big) \\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\!\big(\sum_{k=1}^n x_{1k}x_{kj} - (D_H)_{1j} + x_{1j} \big) \\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\!\big(\sum_{k=1}^n x_{1k}x_{kn} - (D_H)_{1n} + x_{1n} \big)\\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\!\big(\sum_{k=1}^n x_{ik}x_{kj} - (D_H)_{ij} + x_{ij} \big)\\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\!\big(\sum_{k=1}^n x_{nk}x_{kn} - (D_H)_{nn} + x_{nn} \big)
\end{bmatrix}.
\end{equation}
The matrix $D_H$ is diagonal can be represented as $\displaystyle (D_H)_{ii} = \sum_{k} x_{ik}$ where $(D_H)_{ij} = 0$ for $i \neq j$. 
Let $M \in \mathbb{Z}_{> 0}$ be a sufficiently large constant. The known 
$z_{ij}$ represent the terms inside the indicator function such that
\[
g_{ij} = 1 \iff z_{ij} \geq 1 \;\text{and}\; g_{ij} = 0 \iff z_{ij} = 0.
\] The above condition can be modelled using a standard big-$M$ formulation,
and the resulting integer program is given by
\begin{alignat}{3}
\text{Variables: } & \nonumber \\ 
& x_{ij} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,j \in \{1, \dots, n\}, \label{eq::IP_x_variable} \\ 
& y_{ikj} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,k,j \in \{1, \dots, n\}, \label{eq::IP_y_variable} \\
& z_{ij} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,j \in \{1, \dots, n\}. \label{eq::IP_z_variable} \\
\text{Constraints: } & \nonumber \\
\text{(1) Indicator argument: } 
& z_{ii} = \sum_{k = 1}^{n}y_{iki} - \sum_{k = 1}^{n} x_{ik}, 
&& \forall i \in \{1, \dots, n\} , \label{eq::IP_fund_matrix} \\
& z_{ij} = \sum_{k = 1}^{n}y_{ikj} + x_{ij}, 
&& \forall i \neq j \in \{1, \dots, n\} , \label{eq::IP_indicator_arg} \\
\text{(2) Indicator function: } 
& z_{ij} \leq M g_{ij}, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_upper} \\
& z_{ij} \geq g_{ij}, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_lower}\\
\text{(3) Product: } 
& y_{ikj} \leq x_{ik}, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_first} \\
& y_{ikj} \leq x_{kj}, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_second} \\
& y_{ikj} \geq x_{ik} + x_{kj} - 1, 
&& \forall i, k, j \in \{1, \dots, n\} \label{eq::IP_product_lower} \\
\text{(4) Upper Bound: } 
& x_{ij} \leq 1, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_x_upper_bound} \\
& y_{ikj} \leq 1, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_y_upper_bound}. 
\end{alignat} \label{eq::integer_program}

In addition, the adjacency matrix $x$ must satisfy further constraints arising from symmetry and diagonality. Although analogous 
constraints could also be imposed on the auxiliary variables $z$ they are omitted due to redundancy and potentially
introduces artificial degeneracy in the linear program. 
\begin{alignat}{3} 
\text{(5) Symmetry: } 
& x_{ij} = x_{ji}, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::IP_symmetry_x} \\
\text{(6) Diagonal: } 
& x_{ii} = 0, 
&&\qquad \forall i \in \{1, \dots, n\}.  \label{eq::IP_diagonal_x} 
\end{alignat}

\begin{proposition} \label{prop::IP_gives_root}
Let $G$ be a graph and consider the integer program whose variables $x_{ij} \in \{0,1\}$ encode a graph 
$H$ with adjacency matrix $A_H = (x_{ij})$. Then $G$ has a square root $H$ if and only if the integer 
program has a feasible solution $x$ such that the corresponding graph $H$ satisfies $H^2 = G$.
\end{proposition}
\begin{proof}
($\Rightarrow$) Suppose $G$ has a square root $H$ and let $A_H$ be the adjacency matrix of $H$.  $x_{ij} = (A_H)_{ij}$.
% We show that $x$ satisfies all the constraints of the integer program.
Construct the variables $x_{ij}, y_{ikj}, z_{ij}$ and the constant $g_{ij}$ as such 
\begin{enumerate}[label=\roman*)]
  \item Let $x_{ij} = 1$ if $ij \in E_H$ and $=0$ otherwise. 
  \item Let $y_{ikj} = x_{ik}x_{kj}$ for all $i,k,j$. 
  \item Let $\displaystyle z_{ij} = \sum_{k = 1}^{n} y_{ikj} + x_{ij} $ for all $i,j$. 
  \item Let $\displaystyle z_{ii} = \sum_{k  = 1}^{n} y_{iki} - \sum_{k = 1}^{n} x_{ik} =  \sum_{k = 1}^{n} x_{ik}^2 -  \sum_{k = 1}^{n} x_{ik} = 0 $
  since $x_{ij} \in \{0 , 1\} \implies x_{ik}^2 = x_{ik}$. 
  \item Let $g_{ij} = 1$ if $ij \in E_G$ and $=0$ otherwise. 
\end{enumerate}
Verifying constraints of the integer program for the variables 
\begin{enumerate}[label=\roman*)]
  \item As $x_{ij} \in \{0, 1 \} \implies y_{ikj}  \in \{0, 1 \} $, Equations~\eqref{eq::IP_x_variable}-\eqref{eq::IP_z_variable}
  and Equations~\eqref{eq::IP_x_upper_bound}-\eqref{eq::IP_y_upper_bound} are satisfied.  
  \item As $H$ is a simple directed graph, its adjacency matrix is hallow and symmetric such that Equations~\eqref{eq::IP_symmetry_x}
  -\eqref{eq::IP_diagonal_x} are satisfied.  
  \item As $y_{ikj} = x_{ik}x_{kj}$ and all variables are binary, Equations~\eqref{eq::IP_product_upper_first}-
  \eqref{eq::IP_product_lower} are satisfied. 
  \item Equation~\eqref{eq::IP_indicator_arg} by definition is satisfied. 
  \item As $\displaystyle z_{ii} = \sum_{k = 1}^{n} x_{ik}x_{ki} - \sum_{k = 1}^{n} x_{ik}$, Equation~\eqref{eq::IP_fund_matrix} is satisfied. 
  \item As $\displaystyle z_{ij} \geq 1 \iff x_{ij} = 1 \; \text{or} \; \exists k \text{ such that } x_{ik}=x_{kj}= 1$ and as $H^2 = G$,
  have $g_{ij} = 1 \iff 1 \leq d_H(i,j) \leq 2$. Thus $g_{ij} = 1 \iff z_{ij}geq 1$ so the indicator 
  constraints Equations~\eqref{eq::IP_indicator_upper}-\eqref{eq::IP_indicator_lower} hold. 
\end{enumerate}
All constraints of the integer program are satisfied by the constructed variables. Hence, the integer program is feasible.

($\Leftarrow$) Suppose the integer program on graph $G$ has a feasible solution $x$, and let $x_{ij}$ define a graph $H$ by 
\[
ij \in E_H \iff x_{ij} = 1. 
\]
% We show that $H^2 = G$. 
As $x_{ij}$ satisfies the constraints of the integer program, 
\begin{enumerate}[label=\roman*)]
  \item Equation~\eqref{eq::IP_x_variable} and Equation~\eqref{eq::IP_x_upper_bound} gives $x_{ij} \in \{0 , 1\}$ for all $i,j$,
  \item Equation~\eqref{eq::IP_symmetry_x} and Equation~\eqref{eq::IP_diagonal_x} give $x_{ij} = x_{ji}$ and $x_{ii} = 0$,
\end{enumerate}
such that $x_{ij}$ defines a hollow, symmetric matrix with all elements being either $0$ or $1$ which is a valid adjacency matrix of a graph. 
Examining the auxiliary variables $y_{ikj}$ and $z_{ij}$, 
\begin{enumerate}[label=\roman*)]
  \item Equations~\eqref{eq::IP_product_upper_first}-\eqref{eq::IP_product_lower} give the strict equality $y_{ikj} = x_{ik}x_{kj}$ for all $i,j,k$,
  \item Equation~\eqref{eq::IP_indicator_arg} gives $\displaystyle z_{ij} = \sum_{k = 1}^{n}y_{ikj} + x_{ij} = \sum_{k = 1}^{n}x_{ik}x_{kj} + x_{ij}$ for all $i \neq j$,
  \item Equation~\eqref{eq::IP_fund_matrix} gives $\displaystyle z_{ii} = \sum_{k = 1}^{n}y_{iki} + \sum_{n}^{k = 1}x_{ik} = \sum_{k = 1}^{n} x_{ik}x_{ki} - \sum_{n}^{k = 1}x_{ik} = 0$ for all $i$ since
  \[
   \sum_{k = 1}^{n} x_{ik}x_{ki} = \sum_{k = 1}^{n} x_{ik}^2 = \sum_{k = 1}^{n} x_{ik},
  \]
  by symmetry and since $x_{ij} \in \{0 , 1\}$. 
\end{enumerate} 
The indicator constraints Equations~\eqref{eq::IP_indicator_upper}-\eqref{eq::IP_indicator_lower} thus give for $i \neq j$
\[
g_{ij} = 1 \iff \sum_{k=1}^{n} x_{ik}x_{kj} + x_{ij} \geq 1 \; \text{and} \; g_{ij} = 0 \iff \sum_{k=1}^{n} x_{ik}x_{kj} + x_{ij} = 0. 
\]
Since $x_{ij} \in \{0,1\}$, this is equivalent to 
\[
g_{ij} = 1 \iff \Big(x_{ij} = 1 \Big) \lor \Big(\exists k \;\text{such that}\; x_{ik}= 1 \land x_{kj}=1\Big) \; \text{and} \; g_{ij} = 0 \iff x_{ij} = 0. 
\]
By definition of $H$ this means that $g_{ij} = 1 \iff \Big( ij \in E_H \Big) \lor \Big( \exists k \;\text{such that}\; ik,kj \in E_H\Big)$ which is equivalent to 
\[
g_{ij} = 1 \iff 1 \leq d_H(i,j) \leq 2. 
\]
Now apply Lemma~\ref{lemma::G_equal_to_H^2} gives $ij \in E_G \iff ij \in E_H^2$ such that $G=H^2$. 
\end{proof}

\subsection{Linear Program} \label{subsec::LP}

The linear relaxation of the integer program defined by Equations~\eqref{eq::IP_x_variable}--\eqref{eq::IP_diagonal_x} is 
constructed in order to generate a feasible weighted solution in polynomial time algorithm. It is done with identical constraints,
however the decision variables \eqref{eq::IP_x_variable}, \eqref{eq::IP_y_variable} and \eqref{eq::IP_z_variable}
are replaced with
\begin{alignat}{3}
& x_{ij} \in  \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::LP_x_variable} \\
& y_{ikj} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,k,j \in \{1, \dots, n\}, \label{eq::LP_y_variable} \\
& z_{ij} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\} \label{eq::LP_z_variable}. 
\end{alignat} 

The results from the linear programming can give \emph{integral solutions} and \emph{fractional solutions} when the root exists 
and \emph{fractional solutions} and \emph{infeasible solutions} when the root does not exist such that the results in 
Figure~\ref{fig::linear_program_diagram} can be generated. 

The linear relaxation potentially breaks equations in the integer program which are critical to the definition. 

Consider Equations~\eqref{eq::IP_product_upper_first}-\eqref{eq::IP_product_lower} which gives $y_{ikj} = x_{ik}x_{kj}$. When $x$ is binary, this construction of $y$
leads to $y$ being binary as well.  

Consider Equation~\eqref{eq::IP_fund_matrix}, where $\displaystyle z_{ii} = \sum_{k = 1}^{n} y_{iki} - \sum_{k = 1}^{n} x_{ik} + x_{ii}$ can be simplified to 
$\displaystyle z_{ii} = \sum_{k = 1}^{n} x_{ik}^2 - \sum_{k = 1}^{n} x_{ik}$ and is equal to $0$ when $x$ is binary. 
In the linear relaxation this strict inequality no longer holds for fractional values of $x_{ik}$. However, when $x_{ik}$ is fractional
$x_{ik}^2 \leq x_{ik}$ such that $\displaystyle z_{ii} = \sum_{k = 1}^{n} x_{ik}^2 - x_{ik} \leq 0$. As $z \in \mathbb{Z}_{\geq 0}$, this gives 
that $z_{ii} = 0$ as required - no additional modifications are required.

Consider Equation~\eqref{eq::IP_indicator_arg}, where $\displaystyle z_{ij} = \sum_{k = 1}^{n} x_{ik}x_{kj} + x_{ij}$. When $z_{ij} = 0$, 
the equation follows the standard definition of graph squares such that $x_{ij} = 0 $ and for all intermediary $k$s

\subsection{Max Objective Function} \label{subsec::max_obj}

Consider the addition of an objective function maximizing the total weights on all edges to the linear relaxation 
defined by Equations~\eqref{eq::IP_fund_matrix}--\eqref{eq::LP_z_variable} in Section~\ref{subsec::LP} 
\begin{equation} \label{eq::max_obj_function}

\end{equation}
This objective encourages dense solutions and therefore tends to produce relaxations that are more dense and hence more informative

promotes direct edges and reduces the need for fractional paths of length $2$ to satisfy Equation~\eqref{eq::IP_indicator_arg}. As a 
result, the relaxation is more likely to yield integral solutions, as illustrated in Figure~\ref{fig::max_objective_function_diagram}.

complete graph $K_3$ has four roots: itself and three path graphs of length $2$, which are isomorphic to one another. This illustrates 
that the maximization objective preserves consistency up to isomorphism
while still distinguishing structurally meaningful solutions.

\section{Trees}

\citeauthor{ross_1960_square} in 1960 gave a polynomial algorithm for constructing a tree root by using characterizations of the square graph 
to conclude that existing tree square roots are unique up to isomorphism \cite{ross_1960_square}. Using adjacency matrices, we show that the 
modified linear program still gives integral or fractional solutions for graphs with roots and gives fractional or infeasible solutions for graphs 
without roots but always return valid integral root solution if there exists a tree root.

\begin{definition} \label{def::pendant}
  A pendant subgraph is the induced subgraph on three vertices $s,u,v$ where $s$ is a leaf with unique neighbour $u$, and $u$ is a 
  stem of degree $2$ with neighbours $s$ and $v$, where $v$ connects the subgraph to the remainder of the tree.
\end{definition}

\begin{definition} \label{def::leaf_clique}
  A leaf clique subgraph is the induced subgraph on four vertices $s_1,s_2,u,v$ where $s_1,s_2$ are leaves with unique neighbour $u$, and $u$ 
  is an internal vertex of degree $\geq 3$ where $\{s_1,s_2, v\} \subseteq N(u)$, where $v$ connects the subgraph to the remainder of the tree.
\end{definition}

We prove that there exists a polynomial time linear program by showing that 
\begin{enumerate}[label=(\Roman*)]
  \item Every tree contains a pendant or leaf clique subgraph 
  \item Every pendant can be completely classified by single triangular decomposition such that its connection to the remainder of the 
  tree is correctly identified 
  \item Every leaf clique can be completely classified by tetrahedron decomposition such that its connection to the remainder of the 
  tree is correctly identified 
\end{enumerate}
Since any subgraph of a tree is also a tree, using the above three propositions we can coalesce pendants and leaf cliques into fixed
edge assignments such that the maximal linear program pushes the tree solution towards integrality.

\subsection{Single Triangular Decomposition} \label{subsec::single_triangular_decomposition}

Using the local triangular structure of the graph, certain edge assignments in a square root can be fixed. 
For a vertex $s \in V$, define
\[
T(s) = \bigl\{ \{u,v\} \subseteq V \setminus \{s\} \vert (s,u)\in E,\ (s,v)\in E,\ (u,v)\in E \bigr\}
\]
to be the set of $3$-cliques or triangles in $G$ that contain $s$. If $|T(s)| = 1$ where $T(s) = \{\{u,v\}\}$ then the vertex 
$s$ exists in a unique triangle of $G$. When the neighbours have different degree 
$\deg_G(u) \neq \deg_G(v)$, then the vertex with smaller degree must be the leaf-adjacent vertex in the square root. If 
$\deg_G(u) = \deg_G(v)$, then either $u,v$ can be chosen since they are connected to a clique. Importantly, 
if $u,v$ is fixed by equality then it cannot be fixed again in an alternative single triangular decomposition. 
This is assigned deterministically using 
\begin{equation} \label{eq::single_triangle_fix}
x_{sa} = x_{as} = x_{ab} = x_{ba} = 1,
\quad
\text{where }
(a,b) =
\begin{cases}
(u,v), & \deg_G(u) \leq \deg_G(v), \\
(v,u), & \deg_G(u) > \deg_G(v).
\end{cases}
\end{equation}

This addition gives the improved results in Figure~\ref{fig::single_triangular_decomposition_diagram} whilst maintaining 
polynomial runtime. 

\begin{lemma} \label{lem::single_triangle_equal_neighbourhood}
  Let $s,u,v$ form a triangle in square $G$ where $s$ only lies in a single clique. If $\deg_G(u)=\deg_G(v)=d\geq 2$, then
  \[
  N_G(u)\setminus\{s\}=N_G(v)\setminus\{s\}.
  \]
  That is, $N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure in $G$.
\end{lemma}
\begin{proof}
Since $s$ lies in a single clique in $G$, in any tree root $H$ the vertex $s$ must be a leaf, and exactly one of $u$ or $v$ is the
internal vertex adjacent to the remainder of the graph. Assume without loss of generality that $u$ is this internal vertex. Any vertex
adjacent to $v$ in $G$ other than $s$ must arise from a path of length at most two through $u$ in $H$, since $v$
has no other connections in the root besides $u$ and $s$. Hence
\[
N_G(v)\setminus\{s\}\subseteq N_G(u)\setminus\{s\}.
\]
Since $\deg_G(u)=\deg_G(v)$, both neighbourhoods have equal cardinality $|N_G(v)| = |N_G(u)|$ such that via the equality 
criterion for finite sets
\[
N_G(v)\setminus\{s\}
=
N_G(u)\setminus\{s\}.
\]
Hence the vertices adjacent to $u$ and $v$ outside of $s$ have identical adjacency structure in $G$, and therefore
$N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure.
\end{proof}

  choose to assign $x_{su}=x_{uv}=1$ in the tree root $H$ without loss of generality.
\end{proposition}
\begin{proof}

identical clique structure in $G$. Therefore exchanging the roles of $u$ and $v$ preserves the square graph $G$.Hence choosing $u$ 
as the internal vertex and assigning

\]
does not alter feasibility, since the alternative choice obtained by exchanging $u$ and $v$ produces an isomorphic tree root

\end{proof}

\subsection{Tetrahedron Decomposition} \label{subsec::tetrahedron_decomposition}
A natural extension of Section~\ref{subsec::single_triangular_decomposition} can be achieved by considering cliques of size greater than 
three in order to categorize tree structure whereby an internal vertex is connected to more than one leaf. Whilst finding a clique 
of size $\geq k$ for fixed $k$ is polynomial, with complexity of order $O(n^k)$ by simply checking all tuples of size $k$, the decision 
problem of whether a graph $G$ contains a clique of size $k$, as well as the problem of finding a largest clique, is NP-complete. Thus,
it is necessary to only consider cliques of fixed size $4$ which have polynomial complexity $O(n^4)$. This however is sufficient for 
classifying any \hyperref[def::leaf_clique]{leaf clique} in the root, since any branching vertex with exactly two leaves will generate 
a $4$-clique in the square, whilst a branching vertex with more than two leaves will generate multiple $4$-cliques of which one can be 
chosen by arbitrarily selecting any two leaves.

Consider vertices $s_1,s_2,u,v\in V$ such that $K_4 = \{s_1,s_2,u,v\}$ induces a $4$-clique in $G$ and for each leaf $s_1, s_2$ of $u$
their neighbourhoods are equivalent and give the common neighbourhood $N_L = N_G(s_1)\setminus K_4 =N_G(s_2) \setminus K_4$. If
\begin{equation} \label{eq::tetrahedron_neighbourhood_assignment}
\big( N_G(s_1)\setminus K_4 \big)
=
\big( N_G(s_2)\setminus K_4 \big)
\subset
\big( N_G(u)\setminus K_4 \big)
\subseteq
\big( N_G(v)\setminus K_4 \big),
\end{equation}
and if every vertex in the common neighbourhood of each leaf has no neighbours outside the tetrahedron and that neighbourhood itself,
\begin{equation} \label{eq::tetrahedron_leaf_class}
N_G(w)
\subseteq
K_4 \cup N_L
\qquad
\forall\,w\in N_L,
\end{equation}
then $u$ is designated as the internal vertex, $v$ as the external vertex, and $s_1,s_2$ as the leaf vertices of the tetrahedron.


Under Equations~\eqref{eq::tetrahedron_neighbourhood_assignment} and~\eqref{eq::tetrahedron_leaf_class}, the vertices $s_1$ and $s_2$ 
correspond to leaves adjacent to a common internal vertex $u$, whilst $v$ acts as the connecting vertex between $u$ and the remainder
 of the graph. The vertices with identical minimal neighbourhood structure correspond to leaves adjacent to a common 
internal vertex $u$, whilst the vertex $v$, having maximal neighbourhood containment, acts as the connecting vertex between $u$ and the 
remainder of the graph.

For $n=4$, the vertex $u$ is a binary branching vertex contained in a unique $4$-clique. In particular, when $n=5$, the branching vertex $u$ 
has three leaf children $s_1,s_2,s_3$, producing $\displaystyle {3 \choose 2} =3$ distinct $4$-cliques
\[
\{s_1,s_2,u,v\},\quad
\{s_1,s_3,u,v\},\quad
\{s_2,s_3,u,v\}.
\]

More generally, suppose $u$ has $k$ leaf children $s_1,\dots,s_k$, and that these vertices together with $v$ induce a $(k+2)$-clique. 
Then every pair of leaves $(s_i,s_j)$ determines a distinct $4$-clique $\{s_i,s_j,u,v\}$ where the total number of such $4$-cliques generated 
by this branching configuration is $ \displaystyle {k \choose 2}$. Suppose that $s_1,s_2$ are chosen without loss of generality, then 
if the $4$-clique satisfies 
Equation~\eqref{eq::tetrahedron_neighbourhood_assignment} and Equation~\eqref{eq::tetrahedron_leaf_class} can deterministically assign 
values to the edges using  
\begin{equation} \label{eq:tetrahedron_fix}
x_{s_1u}=x_{us_1}
=
x_{s_2u}=x_{us_2}
=
x_{uv}=x_{vu}
=
1,
\end{equation}

  Let $s_1,s_2,u,v$ form a $4$-clique in square $G$. If $N_G(s_1)\setminus\{s_1,s_2,u,v\}=N_G(s_2)\setminus\{s_1,s_2,u,v\}$, then

  N_G(u)\setminus\{s_1, s_2\}=N_G(v)\setminus\{s_1, s_2\}.

\[
N_G(s_1)\setminus\{s_1,s_2,u,v\}
=
N_G(s_2)\setminus\{s_1,s_2,u,v\},

$u$. Consequently $x$ is also at distance at most two from $u$ in

x\in N_G(u)\setminus\{s_1,s_2\},
\]
which implies that $
N_G(v)\setminus\{s_1,s_2\}
\subseteq
N_G(u)\setminus\{s_1,s_2\}$. By symmetry, interchanging the roles of $u$ and $v$ gives
\[
N_G(u)\setminus\{s_1,s_2\}
\subseteq
N_G(v)\setminus\{s_1,s_2\}.
\]
Hence
\[
N_G(u)\setminus\{s_1,s_2\}

N_G(v)\setminus\{s_1,s_2\}.
\]

Therefore the vertices adjacent to $u$ and $v$ outside of

$N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique

\begin{proposition} \label{prop::tetrahedron_fixing}

Now consider any remaining leaf $s_\ell$ adjacent to $u$. Since every leaf adjacent to $u$ lies at distance two from $s_\ell$

\[
N_G(s_\ell)
=
\{u,v,s_1,\dots,s_{\ell-1},s_{\ell+1},\dots,s_k\}.
\]

\[
N_G(s_\ell)
\neq
\{u,v,s_1,\dots,s_{\ell-1},s_{\ell+1},\dots,s_k\}.
\]

\begin{theorem} \label{prop::single_triangular_decomposition+tetrahedron_decomposition}
  If a square graph $G$ has a root $H$ that is a tree, then the single triangular decomposition and tetrahedron decomposition 
  linear program together return an integral solution. 
\end{theorem} 
\begin{proof} % Need to rewrite proof in terms of $G$ rather than $H$ 
Assume for contradiction that there exists a square graph $G$ with a tree root $H$ such that the linear 
program has an optimal, maximal solution $x^*$ that is fractional. Then there exists at least one edge 
$ij \in E_G$ such that $0 < x^*_{ij} < 1 $. Consider the root solution $x^*$ and the following cases. 

Let $i$ be a leaf, $j$ be its neighbour with degree $2$ and let $k \neq i$ be the other neighbour of $j$. Let 
$ij$ be the fractional edge along the path $i - j - k$ which meets the rest of the graph only at $k$. In $G$, $ijk$ 
form a clique in $G$ which by construction is also the only triangle that $i$ can be in. As the linear program applies 
\eqref{eq::single_triangle_fix}, $ij$ pendant edge will be fixed and hence cannot be integral, hence this case 
cannot occur. 

Let $i$ be a leaf, $j$ be its neighbour with degree $\geq 2$ and let $k \neq i$ be another leaf of $j$. Let 
$j$ meet the rest of the graph. Then $ijk$ form a clique and solution $x^*$ can be extended by simply adding an 
additional edge $ik$ to form a root of larger size. This contradicts the maximality of $x^*$. 

The remaining case is where both $i$ and $j$ are internal vertices. Denote the subgraph of $x^*$ from $i$
as $H_i$ and the subgraph from $j$ as $H_j$. Since $x^*$ must be a tree, there are no cycles in $H$ and hence
$H_i$ and $H_j$ are disjoint with only one fractional bridge $ij$ connecting them. Consider $H_i, H_j$ and recursively
apply the above logic such that we are eventually left with at least two subgraphs $H_0, H_1$ that do not have 
fractional bridges and are comprised completely of pendant vertices and leafs. From above two cases, all edges 
in the roots of $H_0, H_1$ are integral, and since the situation where a vertex has multiple leaves can not occur (
there must be pendant 2-leaves together with leaves else would have above case), this completely categorizes the position 
of the node which connects to the rest of the graph such that there are no other automorisphsm which form the same square. 
As this holds for both $H_0$ and $H_1$, the endpoints of the bridge between them can be completely identified such that 
their bridge will be made integral. Applying this inductively thus means that $H_i$ and $H_j$ are completely integral 
such that $i$ and $j$ are identified and $ij$ is made integral. This contradicts the fractional nature of $x^*_{ij}$. 

Since no fractional, solution exists, the optimal, maximal output of the linear program must be completely integral 
if it results in a tree root. 
\end{proof}

\subsection{Vertex Deletion} \label{subsec::vertex_deletion}

As single triangular decomposition gives perfect characterizations of the root using the linear program

\begin{enumerate}[label=\roman*)]

  \item $v$ is connected to the subgraph $G'$ by two edges to vertices $u, w \in G'$ where 
  $uw \notin E_H$ and there is a path of length two between $u,w$ in $G'$. 
\end{enumerate}
where the remaining possibilities have the above sub-cases. 
\begin{enumerate}[label=\roman*)]
  \item If $G^-$ has a valid square root $H'$, then $H = H' + \{v\}$ where $E_H = E_H' \cup vu$. 
  \item If $G'$ has a valid square root $H'$, then $H = H' + \{v\}$ where $E_H = E_H' \cup vu$ or
  $E_H = E_H' \cup vw$ ($E_H = E_H' \cup vu, vw$ also satisfies but is not necessary) 
  \item 
\end{enumerate}

If the solution of the original linear program is fractional, starting from the vertex $v \in V$ with the smallest 
degree, the linear program is ran on the induced subgraph $G' = V - \{v\}$. If the resulting solution is integral, then the 
fractional behaviour of $G$ is localized at $v$ and the algorithm attempts to lift the integral structure of 
$G'$ back to $G$ by enforcing that $v$ is attached to one of its neighbours. For each 
$u \in N_G(v)$, the constraint
\begin{equation}
x_{vu} = 1, \quad x_{uv} = 1,
\label{eq:recursive_attachment}
\end{equation}
is added. If the resulting solution is integral, then $v$ is identified as a leaf vertex adjacent 
to $u$ in the square root, and the solution is accepted.


If no such $u$ yields an integral solution, the process is repeated over all vertices. This technique runs in polynomial 
time, as it performs at most $n$ vertex-deletion checks and, for each such vertex $v$, at most $\deg_G(v) \leq n-1$ constrained 
linear program solves. Since each linear program has $O(n^2)$ variables and constraints and can be solved in polynomial 
time $T_{\mathrm{LP}}(n)$, the overall procedure requires at most $O(n^2 \cdot T_{\mathrm{LP}}(n))$ time, 
and hence remains polynomial in $n$.

\subsection{Dual Problem} \label{subsec::dual_prob} 

We now consider the dual formulation of the max objective linear program defined in Section~\ref{subsubsec::max_obj}
by first restating it in canonical form.

\begin{alignat}{3}
\text{Objective Function: } & \nonumber \\
& \min - \sum_{i=1}^{n} \sum_{j=1}^{n} x_{ij}. \label{eq::primal_max_obj_function}
\end{alignat}
\begin{alignat}{3}
\text{Variables: } & \nonumber \\ 
& x_{ij} \in \mathbb{R}_{\geq 0} \, \Big([0,1]\Big), 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::standard_LP_x_variable} \\
& y_{ikj} \in \mathbb{R}_{\geq 0} \, \Big([0,1]\Big), 
&&\qquad \forall i,k,j \in \{1, \dots, n\}, \label{eq::standard_LP_y_variable} \\
& z_{ij} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\} \label{eq::standard_LP_z_variable}. \\
\text{Constraints: } & \nonumber \\
\text{(1) Indicator argument: } 
& z_{ii} - \sum_{k = 1}^{n}y_{iki} + \sum_{k = 1}^{n} x_{ik} - x_{ii} = 0, 
&& \forall i \in \{1, \dots, n\} , \label{eq::standard_IP_fund_matrix} \\
& z_{ij} - \sum_{k = 1}^{n}y_{ikj} - x_{ij} = 0, 
&& \forall i \neq j \in \{1, \dots, n\} , \label{eq::standard_IP_indicator_arg} \\
\text{(2) Indicator function: } 
& z_{ij} - M g_{ij} \leq 0, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::standard_IP_indicator_upper} \\
& - z_{ij} + g_{ij} \leq 0, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::standard_IP_indicator_lower}\\
\text{(3) Product: } 
& y_{ikj} - x_{ik} \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::standard_IP_product_upper_first} \\
& y_{ikj} - x_{kj} \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::standard_IP_product_upper_second} \\
&  x_{ik} + x_{kj} - y_{ikj} - 1 \leq 0, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::standard_IP_product_lower} \\
\text{(4) Symmetry: } 
& x_{ij} - x_{ji} = 0, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::standard_IP_symmetry_x} \\
\text{(5) Diagonal: } 
& x_{ii} = 0, 
&&\qquad \forall i \in \{1, \dots, n\}, \label{eq::standard_IP_diagonal_x} \\
\text{(6) Variable Bound: } 
& x_{ij} - 1 \leq 0,
&& \forall i, j \in \{1, \dots, n\}, \label{eq::standard_x_upper_bound} \\
& y_{ikj} - 1 \leq 0,
&& \forall i,k, j \in \{1, \dots, n\} \label{eq::standard_y_upper_bound}.
\end{alignat} \label{eq::linear_program}

This gives the dual problem with 
\begin{alignat}{3}
\text{Objective Function: } & \nonumber \\
\max \quad
&
\sum_{i=1}^n \sum_{j=1}^n
\Bigl(
- M g_{ij}\gamma_{ij}
+ g_{ij}\delta_{ij}
- \lambda_{ij}
\Bigr)
- \sum_{i=1}^n \sum_{k=1}^n \sum_{j=1}^n
\bigl(
\eta_{ikj}
+ \mu_{ikj}
\bigr).
\end{alignat}

\begin{alignat}{3}
\text{Dual Variables: } & \nonumber \\

&& \forall i\in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_fund_matrix})}, 
\label{eq::dual_alpha} \\

&& \forall i \neq j \in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_indicator_arg})},
\label{eq::dual_beta} \\

&& \forall i,j \in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_indicator_lower})},
\label{eq::dual_delta} \\

&& \forall i,k,j \in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_product_upper_first})},
\label{eq::dual_epsilon} \\
& \zeta_{ikj} \ge 0,
&& \forall i,k,j \in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_product_upper_second})},
\label{eq::dual_zeta} \\
& \eta_{ikj} \geq 0,
&& \forall i,k,j \in \{1,\dots,n\},
&& \text{(associated with \eqref{eq::standard_IP_product_lower})}, 
\label{eq::dual_eta} \\
& \theta_{ij} \in \mathbb{R},
&& \forall i,j \in \{1,\dots,n\},
