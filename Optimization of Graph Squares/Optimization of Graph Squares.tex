\begin{abstract}
In this paper, we investigate the matrix properties of graph squares, focusing in particular on the
relationship between the adjacency matrix of a graph and the corresponding adjacency of its square. 
\end{abstract}

\section{Introduction} \label{sec::intro}

The problem of finding the square root of a graph is a well-studied field in both mathematics and computer-science, with specific relation to distributed computing, network design and computational biology. Research in this area has primarily focused around the undirected graph $H$ and its square $G = H^2$ which is obtained by adding an edge between every pair of vertices whose distance in $H$ is at most two. 

Much of the existing research on graph square roots has centered on characterizing when a graph admits such a root and on the complexity of reconstructing $H$ from $G$. In contrast, rather than studying the construction problem directly, we analyze the relationship between the adjacency matrix $A_H$ of the root and the adjacency matrix $A_G$ of the square, and show that the problem can be completely categorized as an integer program. 

We then examine the relaxed linear program which can be evaluated in polynomial time using interior point method, and integrate existing literature on special root structures as a stepping-stone towards generating a true polynomial time algorithm. 

\section{Background} \label{sec::background}
Let the \emph{graph root} $H = (V, E_H)$ be a finite, simple, undirected graph, where $V$ is the vertex set with $|V| = n$, and $E_H \subseteq \{\{u,v\} \mid u,v \in V,\; u \neq v\}$ is the edge set. Denote by $d_H(u,v)$ the \emph{graph distance} defined as the number of edges of the shortest path between vertices $u$ and $v$ in $H$. If no such path exists, $d_H(u,v) = \infty$.

The \emph{square} of the graph $H$, denoted by $H^2 = (V, E_{H^2})$ is the graph on the same vertex set with edge set
\begin{equation} \label{eq::fund_def_square}
  E_{H^2} = \{ uv \mid u \neq v,\; 1 \leq d_H(u,v) \leq 2 \}.
\end{equation}
That is, two distinct vertices are adjacent in $H^2$ if and only if they are connected by a path of length one or two in $H$. This is the standard definition of the square of a graph.

Using an alternative construction, let the \emph{graph square} $G = (V, E_G)$ be an undirected graph defined on the same vertex set $V$. Let $A_H \in \{0,1\}^{n \times n}$ denote the adjacency matrix of $H$, where
\[
(A_H)_{ij} =
\begin{cases}
1 ,& \text{if } ij \in E_H, \\
0 ,& \text{otherwise},
\end{cases}
\]
and let $D_H \in \mathbb{R}^{n \times n}$ denote the degree matrix of $H$, where
\[
(D_H)_{ij} =
\begin{cases}
\deg_H(i) ,& \text{if } i = j, \\
0 ,& \text{otherwise}.
\end{cases}
\]
Let $\mathbbm{1}_{\geq 1} : \mathbb{R}^{n \times n} \to \{0,1\}^{n \times n}$ denote the entrywise \emph{indicator function}, defined by
\[
\mathbbm{1}_{\geq 1}(M)_{ij} =
\begin{cases}
1 ,& \text{if } M_{ij} \geq 1, \\
0 ,& \text{otherwise}.
\end{cases}
\]
Define the adjacency matrix of $G$ using the adjacency and degree matrices of $H$
\begin{equation} \label{eq::fund_adj_eq_undirected}
  A_G = \mathbbm{1}_{\geq 1}\big((A_H)^2 - D_H + A_H\big).
\end{equation}
\begin{lemma} \label{lemma::G_equal_to_H^2}
  Let $G$ and $H$ be graphs with adjacency matrices $A_G$ and $A_H$,respectively. The graph $H$ is a root of $G$ if and only if Equation~\ref{eq::fund_adj_eq_undirected} holds
  \[ 
  A_G = \mathbbm{1}_{\geq 1}\!\big((A_H)^2 - D_H + A_H\big).
  \]
\end{lemma}
\begin{proof} 
  For $i \neq j$, since $D_H$ is diagonal, have $(D_H)_{ij} = 0$ and hence
  \[
  \big((A_H)^2 - D_H + A_H\big)_{ij} = (A_H)^2_{ij} + (A_H)_{ij}.
  \]

  Suppose $ij \in E_G$ such that $(A_H)^2_{ij} + (A_H)_{ij} \geq 1$. This implies that at least one of $(A_H)_{ij} = 1$ or $(A_H)^2_{ij} \geq 1$ must be true. If $(A_H)_{ij} = 1$, then $d_H(i,j) = 1$.  If $(A_H)^2_{ij} \geq 1$, then there exists a vertex $k$ such that $ik, kj \in E_H$, and hence $d_H(i,j) \leq 2$. Therefore $1 \leq d_H(i,j) \leq 2$, and thus $ij \in E_{H^2}$.

  Suppose $ij \in E_{H^2}$ such that $1 \leq d_H(i,j) \leq 2$. $D_H$ is diagonal such that $(D_H)_{ij} = 0$. If $d_H(i,j) = 1$, then $(A_H)_{ij} = 1$. If $d_H(i,j) = 2$, then there exists a vertex $k$ such that $ik, kj \in E_H$, and hence $(A_H)^2_{ij} \geq 1$. In either case, $(A_H)^2_{ij} + (A_H)_{ij} \geq 1$, so $(A_G)_{ij} = 1$, and hence $ij \in E_G$.

  For $i = j$, since all graphs are simple, have $(E_{\cdot})_{ii} = 0$, $(A_{\cdot})_{ii} = 0$ and hence
  \[
  \big((A_H)^2 - D_H + A_H\big)_{ii} = (A_H)^2_{ii} - (D_H)_{ii}.
  \]
  $(A_H)^2_{ii} - (D_H)_{ii} = 0$ is always true since the degree matrix of $H$ always has diagonal values $\displaystyle (D_H)_{ii} = \sum_k x_{ik}$ and each edge from vertex $i$ to its neighbour in $H$ will always create a loop to itself in the square and so $\displaystyle (A_H)^2_{ii} = 
  \sum_{k = 1} x_{ik}x_{ki} = \sum_{k = 1} x_{ik}^2$ by the symmetry of adjacency matrix $A_H$. 
  
  Since $x_{ik} \in \{0, 1\}$, the square of the boolean is identical to the boolean $x_ik$ such that $\displaystyle (A_H)^2_{ii} = \sum_k x_{ik} = (D_H)_{ii}$ and $(A_G)_{ii} = \mathbbm{1}_{\geq 1}\big((A_H)^2 - D_H + A_H\big)$ as required. 

  Combining both gives that $E_G = E_{H^2}$, and so $G = H^2$.  
\end{proof}

The formulation of the adjacency matrix of the graph square in Equation~\eqref{eq::fund_adj_eq_undirected} is very powerful. However, the presence of the indicator function introduces a nonlinearity that prevents the problem from being expressed as a purely quadratic form, thereby limiting the direct applicability of existing results from the quadratic optimization literature.

\begin{lemma} \label{lemma::no_indicator_girth_5}
  If $H$ has girth $5$, then Equation~\eqref{eq::fund_adj_eq_undirected} can be simplified to become 
  \begin{equation} \label{eq::fund_adj_eq_undirected_no_indicator}
    A_G = \big((A_H)^2 - D_H + A_H\big).
  \end{equation}
\end{lemma}
\begin{proof}
  Since the graph root has girth $5$ there are no cycles of length $3$ or $4$ in $H$. Thus for every pair of vertices $i \neq j \in V$ there is either no cycle that contains both $i$ and $j$ or $i$ and $j$ lie on a common cycle of length at least $5$. In all cases $(D_H)_{ij} = 0$. If there is no cycle, then there is at most one path between $i$ and $j$. 
  \begin{enumerate}[label=\roman*)]
    \item If the path is of length $2$, then $(A_H)_{ij} = 1, (A_H)^2_{ij} = 0$ and so $(A_G)_{ij} = 0 - 0 + 1 = 1$.
    \item If the path is of length $3$, then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 1$ and so $(A_G)_{ij} = 1 - 0 - 0 = 1$.
    \item If the path is of length $\geq 3$ or a path does not exist, then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 0$ which implies that $(A_G)_{ij} = 0$.
  \end{enumerate}
  If $i$ and $j$ lie on a common cycle of length at least five then there are two paths between $i,j$
  \begin{enumerate}[label=\roman*)]
    \item If one path is of length $2$, then the other is of length $\geq 4$ and hence $(A_H)_{ij} = 1, (A_H)^2_{ij} = 0$ which implies that $(A_G)_{ij} = 0 - 0 + 1 = 1$.
    \item If one path is of length $3$, then the other is of length $\geq 3$ and hence $(A_H)_{ij} = 0, (A_H)^2_{ij} = 1$ which implies that $(A_G)_{ij} = 1 - 0 - 0 = 1$.
    \item If the paths are both of length $\geq 3$,  then $(A_H)_{ij} = 0, (A_H)^2_{ij} = 0$ which implies that $(A_G)_{ij} = 0$. 
  \end{enumerate}
  If $i = j$, then $(A_H)_{ii} = 0$, $(D_H)_{ii} = \deg_H(i)$, and $(A_H)^2_{ii} = \deg_H(i)$. Thus,
  \[
  (A_H)^2_{ii} - (D_H)_{ii} + (A_H)_{ii} = \deg_H(i) - \deg_H(i) + 0 = 0 = (A_G)_{ii}.
  \]
\end{proof}

\subsection{Integer Program} \label{subsec::IP}
Let $x_{ij} \in \{0, 1\}$ be the decision variables of $A_H$, let $y_{ikj} \in \{0, 1\}$ be the decision variable for $x_{ik}x_{kj}$ in $(A_H)^2_{ij}$ and let $z_{ij} \in \mathbb{Z}_{\geq 0}$ represent the argument inside the indicator function. 

$\displaystyle 
D_H =
\begin{bmatrix}
\deg_H(1) & 0      & \cdots & 0 & 0      \\
0         & \ddots & \vdots & \ddots & \vdots \\
\vdots    & \cdots & \deg_H(i) & \cdots & 0      \\
0         & \ddots & \vdots & \ddots & \vdots \\
0         & \cdots & 0      & \cdots & \deg_H(n)
\end{bmatrix}, 
$ 
$\displaystyle A_H =
\begin{bmatrix}
x_{11} & \cdots & x_{1j} & \cdots & x_{1n} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
x_{i1} & \cdots & x_{ij} & \cdots & x_{in} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
x_{n1} & \cdots & x_{nj} & \cdots & x_{nn}
\end{bmatrix}
$ such that 
\[
(A_H)^2 = \begin{bmatrix}
\displaystyle \sum_{k=1}^n x_{1k}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{1k}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{1k}x_{kn} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
\displaystyle \sum_{k=1}^n x_{ik}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{ik}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{ik}x_{kn} \\
\vdots  & \ddots  & \vdots & \ddots & \vdots  \\
\displaystyle \sum_{k=1}^n x_{nk}x_{k1} & \cdots & \displaystyle \sum_{k=1}^n x_{nk}x_{kj} & \cdots & \displaystyle \sum_{k=1}^n x_{nk}x_{kn}
\end{bmatrix}.
\]
Then with known constants $g_{ij} \in \{0, 1\}$ from $A_G$, we can formulate Equation~\eqref{eq::fund_adj_eq_undirected} as an integer program by equating entries of each matrix to give the following linear system of $n^2$ equations.
\begin{equation}  \label{eq::fund_programming_equation}
\begin{bmatrix}
g_{11} \\
\vdots \\
g_{1j} \\
\vdots \\
g_{1n} \\
\vdots \\
g_{ij} \\
\vdots \\
g_{nn}
\end{bmatrix} 
=
\begin{bmatrix}
\mathbbm{1}_{\geq 1}\big(z_{11}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{1j}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{1n}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{ij}\big) \\
\vdots \\
\mathbbm{1}_{\geq 1}\big(z_{nn}\big)
\end{bmatrix}
=
\begin{bmatrix} \displaystyle
\mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{k1} - (D_H)_{11} + x_{11} \big) \\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{kj} - (D_H)_{1j} + x_{1j} \big) \\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{1k}x_{kn} - (D_H)_{1n} + x_{1n} \big)\\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{ik}x_{kj} - (D_H)_{ij} + x_{ij} \big)\\
\vdots \\ \displaystyle
\mathbbm{1}_{\geq 1}\big(\sum_{k=1}^n x_{nk}x_{kn} - (D_H)_{nn} + x_{nn} \big)
\end{bmatrix}.
\end{equation}
The matrix $D_H$ is diagonal and can be represented as $\displaystyle (D_H)_{ii} = \sum_{k} x_{ik}$ where $(D_H)_{ij} = 0$ for $i \neq j$. The variable $z_{ij}$ represent the terms inside the indicator function such that
\[
g_{ij} = 1 \text{ if and only if } z_{ij} \geq 1 \text{ and } g_{ij} = 0 \text{ if and only if } z_{ij} = 0.
\]  
Let $M \in \mathbb{Z}_{> 0}$ be a sufficiently large constant that may be bounded by the number of edges, that is $M \leq n(n -1) = O(n^2)$. The above condition can be modelled using a standard big-$M$ formulation. The resulting integer program is given by
\begin{alignat}{3}
\text{Objective Function: } & \nonumber \\
\max \quad & 0 \label{eq::integer_objective_function}
\end{alignat}
\begin{alignat}{3}
\text{Variables: } & \nonumber \\ 
& x_{ij} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,j \in \{1, \dots, n\}, \label{eq::IP_x_variable} \\ 
& y_{ikj} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,k,j \in \{1, \dots, n\}, \label{eq::IP_y_variable} \\
& z_{ij} \in \mathbb{Z}_{\geq 0}, 
&& \forall i,j \in \{1, \dots, n\}, \label{eq::IP_z_variable} \\
\text{Constraints: } & \nonumber \\
\text{(1) Indicator argument: } 
& z_{ii} = \sum_{k = 1}^{n}y_{iki} - \sum_{k = 1}^{n} x_{ik}, 
&& \forall i \in \{1, \dots, n\}, \label{eq::IP_fund_matrix} \\
& z_{ij} = \sum_{k = 1}^{n}y_{ikj} + x_{ij}, 
&& \forall i \neq j \in \{1, \dots, n\}, \label{eq::IP_indicator_arg} \\
\text{(2) Indicator function: } 
& z_{ij} \leq M g_{ij}, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_upper} \\
& z_{ij} \geq g_{ij}, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_indicator_lower}\\
\text{(3) Product: } 
& y_{ikj} \leq x_{ik}, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_first} \\
& y_{ikj} \leq x_{kj}, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_upper_second} \\
& y_{ikj} \geq x_{ik} + x_{kj} - 1, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_product_lower} \\
\text{(4) Upper Bound: } 
& x_{ij} \leq 1, 
&& \forall i, j \in \{1, \dots, n\}, \label{eq::IP_x_upper_bound} \\
& y_{ikj} \leq 1, 
&& \forall i, k, j \in \{1, \dots, n\}, \label{eq::IP_y_upper_bound} \\
\text{(5) Symmetry: } 
& x_{ij} = x_{ji}, 
&&\quad \forall i,j \in \{1, \dots, n\}, \label{eq::IP_symmetry_x} \\
\text{(6) Diagonal: } 
& x_{ii} = 0, 
&&\quad \forall i \in \{1, \dots, n\}.  \label{eq::IP_diagonal_x} 
\end{alignat} \label{eq::integer_program}

The constraints in Equations~\eqref{eq::IP_symmetry_x}--\eqref{eq::IP_diagonal_x} impose the required symmetry and diagonal structure on the adjacency matrix $A_H$. Although analogous constraints could also be imposed on the auxiliary variables $z$, they are omitted as they are redundant and may introduce artificial degeneracy into the linear program.

The integer program on the arbitrary graph $G$ using Equations~\eqref{eq::IP_x_variable}--\eqref{eq::IP_diagonal_x} together with the constant objective function Equation~\eqref{eq::integer_objective_function} is denoted as $\mathcal{IP}(G)$. s

\begin{theorem} \label{theorem::IP_gives_root}
  Let $G$ be a graph and consider the integer program whose variables $x_{ij} \in \{0,1\}$ encode a graph $H$ with adjacency matrix $A_H = (x_{ij})$. Then $G$ has a square root $H$ if and only if the integer  program has a feasible solution $x$. The feasible solution $x$ defines entries of the adjacency matrix of a graph $H$ that satisfies $H^2 = G$.
\end{theorem}
\begin{proof}
  Suppose $G$ has a square root $H$ and let $A_H$ be the adjacency matrix of $H$ and let  $x_{ij} = (A_H)_{ij}$. Construct the variables $x_{ij}, y_{ikj}, z_{ij}$ and the constant $g_{ij}$ as follows.
  \begin{enumerate}[label=\roman*)]
    \item Let $x_{ij} = 1$ if $ij \in E_H$ and $x_{ij}=0$ otherwise. 
    \item Let $y_{ikj} = x_{ik}x_{kj}$ for all $i,k,j$. 
    \item Let $\displaystyle z_{ij} = \sum_{k = 1}^{n} y_{ikj} + x_{ij} $ for all $i,j$. 
    \item Let $\displaystyle z_{ii} = \sum_{k  = 1}^{n} y_{iki} - \sum_{k = 1}^{n} x_{ik} =  \sum_{k = 1}^{n} x_{ik}^2 -  \sum_{k = 1}^{n} x_{ik} = 0 $
    since $x_{ij} \in \{0 , 1\} \implies x_{ik}^2 = x_{ik}$. 
    \item Let $g_{ij} = 1$ if $ij \in E_G$ and $g_{ij}=0$ otherwise. 
  \end{enumerate}
  Verifying constraints of the integer program for the variables 
  \begin{enumerate}[label=\roman*)]
    \item As $x_{ij} \in \{0, 1 \}$, then $y_{ikj}  \in \{0, 1 \}$, Equations~\eqref{eq::IP_x_variable}-\eqref{eq::IP_z_variable}
    and Equations~\eqref{eq::IP_x_upper_bound}-\eqref{eq::IP_y_upper_bound} are satisfied.  
    \item As $H$ is a simple directed graph, its adjacency matrix has zero diagonal and is symmetric. Therefore Equations~\eqref{eq::IP_symmetry_x}
    -\eqref{eq::IP_diagonal_x} are satisfied.  
    \item As $y_{ikj} = x_{ik}x_{kj}$ and all variables are binary. Therefore Equations~\eqref{eq::IP_product_upper_first}-
    \eqref{eq::IP_product_lower} are satisfied. 
    \item Equation~\eqref{eq::IP_indicator_arg} by definition is satisfied. 
    \item As $\displaystyle z_{ii} = \sum_{k = 1}^{n} x_{ik}x_{ki} - \sum_{k = 1}^{n} x_{ik}$, Equation~\eqref{eq::IP_fund_matrix} is satisfied. 
    \item As $\displaystyle z_{ij} \geq 1 \text{ if and only if } x_{ij} = 1 \text{ or there exists } k \text{ such that } x_{ik}=x_{kj}= 1$ and as $H^2 = G$, have $g_{ij} = 1 \text{ if and only if } 1 \leq d_H(i,j) \leq 2$. Thus $g_{ij} = 1 \text{ if and only if } z_{ij} \geq 1$. Therefore indicator constraints Equations~\eqref{eq::IP_indicator_upper}-\eqref{eq::IP_indicator_lower} hold. 
  \end{enumerate}
  All constraints of the integer program are satisfied by the constructed variables. Hence, the integer program is feasible.

  Suppose the integer program on graph $G$ has a feasible solution $x$, and let $x_{ij}$ define a graph $H$ by 
  \[
  ij \in E_H \text{ if and only if } x_{ij} = 1. 
  \]
  As $x_{ij}$ satisfies the constraints of the integer program, 
  \begin{enumerate}[label=\roman*)]
    \item Equation~\eqref{eq::IP_x_variable} and Equation~\eqref{eq::IP_x_upper_bound} gives $x_{ij} \in \{0 , 1\}$ for all $i,j$,
    \item Equation~\eqref{eq::IP_symmetry_x} and Equation~\eqref{eq::IP_diagonal_x} give $x_{ij} = x_{ji}$ and $x_{ii} = 0$,
  \end{enumerate}
  such that $x_{ij}$ defines a zero diagonal, symmetric matrix with all elements being either $0$ or $1$ which is a valid adjacency matrix of a graph. 
  Examining the auxiliary variables $y_{ikj}$ and $z_{ij}$, 
  \begin{enumerate}[label=\roman*)]
    \item Equations~\eqref{eq::IP_product_upper_first}-\eqref{eq::IP_product_lower} give the strict equality $y_{ikj} = x_{ik}x_{kj}$ for all $i,j,k$,
    \item Equation~\eqref{eq::IP_indicator_arg} gives $\displaystyle z_{ij} = \sum_{k = 1}^{n}y_{ikj} + x_{ij} = \sum_{k = 1}^{n}x_{ik}x_{kj} + x_{ij}$ for all $i \neq j$,
    \item Equation~\eqref{eq::IP_fund_matrix} gives $\displaystyle z_{ii} = \sum_{k = 1}^{n}y_{iki} + \sum_{n}^{k = 1}x_{ik} = \sum_{k = 1}^{n} x_{ik}x_{ki} - \sum_{n}^{k = 1}x_{ik} = 0$ for all $i$ since
    \[
    \sum_{k = 1}^{n} x_{ik}x_{ki} = \sum_{k = 1}^{n} x_{ik}^2 = \sum_{k = 1}^{n} x_{ik},
    \]
    by symmetry and since $x_{ij} \in \{0 , 1\}$. 
  \end{enumerate} 
  The indicator constraints Equations~\eqref{eq::IP_indicator_upper}-\eqref{eq::IP_indicator_lower} thus give for $i \neq j$
  \[
  g_{ij} = 1 \text{ if and only if } \sum_{k=1}^{n} x_{ik}x_{kj} + x_{ij} \geq 1  \text{ and }  g_{ij} = 0 \text{ if and only if } \sum_{k=1}^{n} x_{ik}x_{kj} + x_{ij} = 0. 
  \]
  Since $x_{ij} \in \{0,1\}$, this is equivalent to saying that $
  g_{ij} = 1$ if and only if $\Big(x_{ij} = 1 \Big) \lor \Big(\text{ there exists } k $ \newline
  $\text{ such that } x_{ik}= 1 \land x_{kj}=1\Big)$ and  $g_{ij} = 0$ if and only if $x_{ij} = 0$. 
  By definition of $H$ this means that $g_{ij} = 1 \text{ if and only if } \Big( ij \in E_H \Big) \lor \Big( \text{ there exists } k \text{ such that } ik,kj \in E_H\Big)$ which is equivalent to 
  \[
  g_{ij} = 1 \text{ if and only if } 1 \leq d_H(i,j) \leq 2. 
  \]
  Applying Lemma~\ref{lemma::G_equal_to_H^2} thus gives $ij \in E_G \text{ if and only if } ij \in E_H^2$. Therefore $G=H^2$. 
\end{proof}

\subsection{Linear Relaxation} \label{subsec::LP}

We construct the linear relaxation of the integer program defined by Equations~\eqref{eq::integer_objective_function}--\eqref{eq::IP_diagonal_x}, which generates a feasible weighted solution in polynomial time. The relaxation has identical constraints and the same constant objective function, however the decision variables Equations~\eqref{eq::IP_x_variable}--\eqref{eq::IP_z_variable} are replaced with
\begin{alignat}{3}
& x_{ij} \in  \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\}, \label{eq::LP_x_variable} \\
& y_{ikj} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,k,j \in \{1, \dots, n\}, \label{eq::LP_y_variable} \\
& z_{ij} \in \mathbb{R}_{\geq 0}, 
&&\qquad \forall i,j \in \{1, \dots, n\} \label{eq::LP_z_variable}. 
\end{alignat}

The following proposition establishes that whenever a graph root exists, the linear relaxation will never return an infeasible solution. 

\begin{proposition} \label{prop::linear_relaxation_feasibility}
  If $G$ has a square root $H$, then $\mathcal{LP}(G)$ is feasible. 
\end{proposition}

\begin{proof}
  Suppose that $G$ has a square root $H$. By Theorem~\ref{theorem::IP_gives_root}, the adjacency matrix $A_H$ gives a feasible solution to $\mathcal{IP}(G)$. Since $\mathcal{LP}(G)$ is obtained by relaxing the integrality constraints of $\mathcal{IP}(G)$, every feasible solution of $\mathcal{IP}(G)$ is also feasible for $\mathcal{LP}(G)$. Hence $\mathcal{LP}(G)$ is feasible.
\end{proof}

\section{Linear Program} \label{sec::LP}
In the integer program and its linear relaxation, the most important object is the triangle, the $3$-clique that denotes the relationship between any three vertices $i,j,k$. Consider the following scenarios for the square $G$

\begin{enumerate}[label=(\Roman*)]
  \item \label{lp::single_edge} $ij$ is the only edge for $i,j,k$ in $G$. There are three situations which can occur  
  \begin{enumerate}[label=(\roman*)]
    \item \label{lp::single_edge_no_vertex} $ij$ edge is present in root $H$ and there does not exist another vertex $l$ where $il, lj$ in $H$
    \item \label{lp::no_edge_vertex} $ij$ edge is not present in root $H$ and there exists another vertex $l$ where $il, lj$ in $H$
    \item \label{lp::edge_vertex} $ij$ edge is present in root $H$ and there exists another vertex $l$ where $il, lj$ in $H$
  \end{enumerate}
  \item \label{lp::three_edges} $i,j,k$ form a clique in $G$ with $ij, jk, ki$ present in $G$. \ref{lp::single_edge} applies for all three 
  edges, with the caveat that any vertex can act as the corresponding $l$ vertex for the other two vertices. 
\end{enumerate}
These common scenarios clearly exhibit the ``hardness'' of the square root problem as each present edge of a triangle branches out into three possibilities. However, the remaining case gives critical structure to the problem 
\begin{enumerate}[resume, label=(\Roman*)]
  \item \label{lp::two_edges} $ij$ and $jk$ are the two edges for $i,j,k$ in $G$. Since $ki$ does not appear in $G$ there must not be an edge in $H$ for both $ij$ and $jk$ in $H$, giving the scenarios for both edges
  \begin{enumerate}[label=(\arabic*)] 
    \item \label{lp::two_edges_single_edge_1} $ij$ edge follows \ref{lp::single_edge_no_vertex} and $jk$ follows \ref{lp::no_edge_vertex}
    \item \label{lp::two_edges_single_edge_2} $ij$ edge follows \ref{lp::no_edge_vertex} and $jk$ follows \ref{lp::single_edge_no_vertex}
    \item \label{lp::two_edges_no_edges} $ij$ edge follows \ref{lp::no_edge_vertex} and $jk$ follows \ref{lp::no_edge_vertex}
  \end{enumerate}
\end{enumerate}
With only three branches for the two edges, the structure of \ref{lp::two_edges} is the most efficient method of analyzing any graph square since it allows an efficient logical disjunction to simply summarize all possible scenarios. 

\begin{lemma} \label{lemma::g_ab=0}
  Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b \in V_G$ such that $ab \notin E_G$. For every feasible solution $x$ of $\mathcal{LP}(G)$ and every $v \in V_G$ with $v \neq a,b$ we have $x_{ab} = y_{avb} = 0$. 
\end{lemma}
\begin{proof}
  Using Equations~\eqref{eq::IP_indicator_upper}--\eqref{eq::IP_indicator_lower} if $g_{ab} = 0$, then $z_{ab} \leq 0$ together with $z_{ab} \geq = 0$ implies that $z_{ab} = 0$. Then Equation~\eqref{eq::IP_indicator_arg} gives that $\displaystyle 0 = \sum_{v = 1}^{n} y_{avb} + x_{ab}$ which only holds when each variable is equal to $0$ since variables of the form $x_{ab}, y_{avb}$ are non-negative. This gives that $x_{ab} = 0$ and $y_{avb} = 0$ for all vertices $v \in V \backslash \{a,b\}$. 
\end{proof}

\begin{lemma} \label{lemma::g_ab=0,two_edge_sum}
  Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b \in V_G$ such that $ab \notin E_G$. For every feasible solution $x$ of $\mathcal{LP}(G)$ and every $v \in V_G$ with $v \neq a,b$ we have
  \[
  x_{av} + x_{vb} \leq 1.
  \]
\end{lemma}
\begin{proof}
  By Lemma~\eqref{lemma::g_ab=0}, $y_{avb}=0$. By Equation~\eqref{eq::IP_indicator_lower}, $0 = y_{avb} \geq x_{av} + x_{vb} - 1$. Rearranging yields $x_{av} + x_{vb} \leq 1$, as required. 
\end{proof}

\begin{lemma} \label{lemma::g_ab=0,x_av=1}
  Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,b,v \in V_G$ such that $ab \notin E_G$. If a feasible solution $x$ of $\mathcal{LP}(G)$ satisfies $x_{av}=1$, then $x_{vb}=0$.
\end{lemma}
\begin{proof}
  Since $ab \notin E_G$, Lemma~\eqref{lemma::g_ab=0,two_edge_sum} gives $x_{av}+x_{vb} \leq 1$. Since $x_{av}=1$, it follows that $1+x_{vb} \leq 1$, and hence $x_{vb} \leq 0$. By the non-negativity of Equation~\eqref{eq::LP_x_variable}, $x_{vb} \geq 0$, and therefore $x_{vb}=0$.
\end{proof}

\begin{lemma}[$0$ Path Propagation] \label{lemma::0_path_propagation}
  Let $G$ and $H$ be graphs such that $G = H^2$. Let $a,i,j,b \in V_G$ such that $a,i,j,b$ form a shortest induced path in $H$ between $a$ and $b$. For every feasible solution $x$ of $\mathcal{LP}(G)$,
  \[
  x_{ai}=1 \implies x_{ib}=0, \quad x_{ia}=1 \implies x_{bi}=0, \quad x_{jb}=1 \implies x_{aj}=0, \quad x_{bj}=1 \implies x_{ja}=0.
  \]
\end{lemma}
\begin{proof}
  Since $a,i,j,b$ form shortest path in $H$, we have $d_H(a,b)=3$ and therefore $ab \notin E_G$. By Lemma~\eqref{lemma::g_ab=0,x_av=1}, setting  $x_{ai}=1$ gives $x_{ib}=0.$ Similarly, setting $x_{jb}=1$ gives $ x_{aj}=0$. The remaining two implications follow by symmetry.
\end{proof}

\begin{lemma}[$1$-Path Propagation] \label{lemma::1_path_propagation}
  Let $G$ be a graph and let $x$ be a feasible solution of $\mathcal{LP}(G)$. Suppose that $u-v-a_1$ form a shortest induced path in $H$ between $u$ and $a$ and that $x_{uv}=1$. Let $A=\{a_1,\ldots,a_r\}$ and $B=\{b_1,\ldots,b_s\}$ be such that every vertex $k \in V_G\setminus\{u,v\}$ satisfies
  $k \in A\cup B$, and suppose that 
  \[
  x_{um}=0 \text{ for every } m \in A, \quad g_{un}=0 \text{ for every } n \in B,
  \]
  and $g_{sa_1}=0$ for the vertex $s$ preceding $u$ on the shortest path to $a_1$. Then
  \[
  x_{va_1}=1.
  \]
\end{lemma}
\begin{proof}
  Without loss of generality consider $a_1\in A$. Since $ua_1\in E_G$, have $g_{ua_1}=1$, and hence Equation~\eqref{eq::IP_indicator_lower} gives $z_{ua_1}\geq1$. Since $x_{ua_1}=0$, Equation~\eqref{eq::IP_indicator_arg} gives
  \[
  1 \leq z_{ua_1} = x_{ua_1}+\sum_{k\in V_G}y_{uka_1} = \sum_{k\in V_G}y_{uka_1}.
  \]
  The terms corresponding to $k=s$ and $k=u$ are zero since $g_{sa_1}=g_{uu}=0$, and hence Lemma~\eqref{lemma::g_ab=0} gives $x_{sa_1}=x_{uu}=0$. Therefore, by Equations~\eqref{eq::IP_product_upper_first}--\eqref{eq::IP_product_lower},
  \[
  y_{usa_1}=y_{uua_1}=0.
  \]
  For every $m\in A$, we have $x_{um}=0$, and hence Equation~\eqref{eq::IP_product_upper_first} gives
  \[
  y_{uma_1}=0.
  \]
  Similarly, for every $n\in B$, we have $g_{un}=0$, and therefore Lemma~\eqref{lemma::g_ab=0} gives $x_{un}=0$. Thus Equation~\eqref{eq::IP_product_upper_first} gives
  \[
  y_{una_1}=0.
  \]
  Thus all terms in the sum are zero except possibly $y_{uva_1}$, giving
  \[
  1 \leq y_{uva_1}.
  \]
  Since $x_{uv}=1$, Equation~\eqref{eq::IP_product_upper_first} gives $y_{uva_1}\leq x_{uv}=1$, while Equation~\eqref{eq::IP_product_upper_second} gives $y_{uva_1} \leq x_{va_1}$. Hence $1 \leq y_{uva_1} \leq x_{va_1} \leq 1$, and therefore
  \[
  x_{va_1}=1.
  \]
\end{proof}

\section{Trees}

\citeauthor{ross_1960_square} in 1960 gave an algorithm for constructing a tree root by using characterizations of the square graph to conclude that existing tree square roots are unique up to isomorphism \cite{ross_1960_square}. Building upon these insights, \citeauthor{lin_1995_square_roots} bypassed the inefficiency of general maximal clique enumeration in \citeauthor{ross_1960_square}'s approach, by developing an optimal $\mathcal{O}(\vert{}V\vert{} + \vert{}E\vert{})$ linear time algorithm to recognize tree squares and reconstruct their unique tree roots\cite{lin_1995_square_roots}.

We show that a modified linear program will always return valid integral root solution if there exists a tree root whilst still remaining feasible for graphs with roots in general. We show that every tree contains one of two structural configurations that can be identified from its square $G$. These configurations allow us to identify a leaf, or a group of leaves, and thereby recover a path $s-u-v$ in $H$, where $s$ is a leaf. Once such a path is identified, we can fix the corresponding edge variables and use the resulting constraints to propagate the assignments through the $\mathcal{LP}$.

\begin{definition}[Sparse Pendant] \label{def::sparse_pendant}
  Let $H$ be a graph. A \textbf{sparse pendant} is a path $s-u-v$ in $H$ such that $s$ is a leaf and $u$ has degree $2$.
\end{definition}
\begin{definition}[Pendant Fork] \label{def::pendant_fork}
  Let $H$ be a graph. A \textbf{pendant fork} is the induced subgraph of $H$ on four vertices $s_1,s_2,u,v$, where $s_1$ and $s_2$ are leaves adjacent to $u$, and every neighbour of $u$ other than $v$ is a leaf.
\end{definition}

\begin{lemma} \label{lemma::tree_pendant_pendant_fork}
  Every tree with at least $3$ vertices contains either a \hyperref[def::sparse_pendant]{sparse pendant} or a \hyperref[def::pendant_fork]{pendant fork} subgraph.
\end{lemma}
\begin{proof} 
  Let $T$ be a rooted tree with $|V_T| \ge 3$, and root $T$ at an arbitrary vertex. Since $T$ is finite, there exists a leaf, say $s$, of maximum depth. Let $u$ be the parent of $s$. Every child of $u$ is a leaf, since if there was some child $w$ of $u$ that is not a leaf, then $w$ would have a child and hence there would exist a vertex at depth strictly greater than that of $s$, contradicting the choice of $s$.

  If $u$ is not the root, let $v$ denote its parent. Every neighbour of $u$ other than $v$ is a child of $u$, and hence is a leaf. If $u$ has exactly one such leaf neighbour, say $s$, then $u$ has exactly two neighbours, namely $s$ and $v$, and hence $\deg_T(u)=2$. Therefore, the induced subgraph on
  \[
  \{s,u,v\}
  \]
  forms a pendant subgraph.

  Otherwise $u$ has at least two leaf neighbours $s_1,s_2$. Choose another neighbour $v$ of $u$ distinct from $s_1,s_2$. If $u$ is not the root, choose $v$ as its parent. Since $v$ is distinct from $s_1$ and $s_2$, have $\{s_1,s_2,v\}\subseteq N(u)$, and $\deg_T(u)\ge3$. Hence the induced subgraph on
  \[
  \{s_1,s_2,u,v\}
  \]
  forms a pendant fork subgraph.

  If $u$ is the root, then all neighbours of $u$ are children and hence are leaves. Since $|V_T| \ge 3$, $u$ has at least two neighbours. If it has exactly two neighbours, the induced subgraph on these three vertices forms a sparse pendant. Otherwise, choose any three distinct children $s_1,s_2,v$ of $u$. Then $\{s_1,s_2,v\}\subseteq N_T(u)$, and the induced subgraph on
  \[
    \{s_1,s_2,u,v\}
  \]
  forms a pendant fork subgraph.
\end{proof}



\subsection{Single Triangular Decomposition} \label{subsec::single_triangular_decomposition}

We first consider how to identify the sparse pendant described in Definition~\ref{def::sparse_pendant} from $G$. For a vertex $s \in V$, define
\[
T(s) = \bigl\{ \{u,v\} \subseteq V \setminus \{s\} \vert (s,u)\in E,\ (s,v)\in E,\ (u,v)\in E \bigr\}
\]
to be the set of $K_3$s in $G$ that contain $s$. If $|T(s)| = 1$ where $T(s) = \{\{u,v\}\}$ then the vertex $s$ exists in a unique $K_3$ of $G$. When the neighbours have different degree $\deg_G(u) \neq \deg_G(v)$, then the vertex with smaller degree must be the stem vertex in the square root. This is assigned deterministically for a single triangle $s$ using
\begin{equation} \label{eq::single_triangle_fix}
  x_{sa} = x_{as} = x_{ab} = x_{ba} = 1, \quad \text{where } (a,b) =
  \begin{cases}
  (u,v), & \deg_G(u) \leq \deg_G(v), \\
  (v,u), & \deg_G(u) > \deg_G(v).
  \end{cases}
\end{equation}

\begin{lemma} \label{lemma::single_triangle_equal_neighbourhood}
  Let $s,u,v$ form a triangle in square $G$ where $s$ only lies in a single clique. If $\deg_G(u)=\deg_G(v)=d\geq 2$, then
  \[
  N_G(u)\setminus\{s\}=N_G(v)\setminus\{s\}.
  \]
  That is, $N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure in $G$.
\end{lemma}
\begin{proof}
  Since $s$ lies in a single clique in $G$, in any tree root $H$ the vertex $s$ must be a leaf, and exactly one of $u$ or $v$ is the internal vertex adjacent to the remainder of the graph. Assume without loss of generality that $u$ is this internal vertex. Any vertex adjacent to $v$ in $G$ other than $s$ must arise from a path of length at most two through $u$ in $H$, since $v$ has no other connections in the root besides $u$ and $s$. Hence
  \[
  N_G(v)\setminus\{s\}\subseteq N_G(u)\setminus\{s\}.
  \]
  Since $\deg_G(u)=\deg_G(v)$, both neighbourhoods have equal cardinality $|N_G(v)| = |N_G(u)|$ such that via the equality criterion for finite sets
  \[
  N_G(v)\setminus\{s\}
  =
  N_G(u)\setminus\{s\}.
  \]
  Hence the vertices adjacent to $u$ and $v$ outside of $s$ have identical adjacency structure in $G$, and therefore $N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure.
\end{proof}

\begin{proposition} \label{prop::single_triangular_decomposition_equality}
  Let $s,u,v$ form a triangle in square $G$ where $s$ only lies in a single clique. If $\deg_G(u)=\deg_G(v)=d\geq2$, then one can choose to assign $x_{su}=x_{uv}=1$ in the tree root $H$ without loss of generality.
\end{proposition}
\begin{proof}
  Since $s$ lies in a single clique in $G$, in the root $H$, $s$ must be a leaf with either $u$ or $v$ being the internal vertex adjacent to the remainder of the graph. Assume without loss of generality that $u$ is the internal vertex.

  By Lemma~\ref{lemma::single_triangle_equal_neighbourhood}, $N_G(u)\setminus\{s\}=N_G(v)\setminus\{s\}$, such that $u$ and $v$ induce identical clique structure in $G$. Therefore exchanging the roles of $u$ and $v$ preserves the square graph $G$.Hence choosing $u$ as the internal vertex and assigning
  \[
  x_{su}=x_{uv}=1,
  \]
  does not alter feasibility, since the alternative choice obtained by exchanging $u$ and $v$ produces an isomorphic tree root with identical square graph.
\end{proof}

\subsection{Tetrahedron Decomposition} \label{subsec::tetrahedron_decomposition}
We now consider how to identify the pendant fork described in Definition~\ref{def::pendant_fork} from $G$. A natural extension of Section~\ref{subsec::single_triangular_decomposition} can be achieved by considering cliques of size greater than three in order to categorize tree structure whereby an internal vertex is connected to more than one leaf. 

Whilst finding a clique of size $\geq k$ for fixed $k$ is polynomial, with complexity of order $O(n^k)$ by simply checking all tuples of size $k$, the decision problem of whether a graph $G$ contains a clique of size $k$, as well as the problem of finding a largest clique, is $\mathbf{NP}$-complete. Thus, it is necessary to consider only cliques of fixed size $4$, for which the problem of determining their existence has polynomial complexity $O(n^4)$. This however is sufficient for classifying any \hyperref[def::pendant_fork]{pendant fork}, since any branching vertex with exactly two leaves will generate a $K_4$ in the square, whilst a branching vertex with more than two leaves will generate multiple $K_4$s of which one can be chosen by arbitrarily selecting any two leaves.

Consider vertices $s_1,s_2,u,v\in V$ such that $K_4 = \{s_1,s_2,u,v\}$ induces a $4$-clique in $G$ and for each leaf $s_1, s_2$ of $u$ their neighbourhoods are equivalent and give the common neighbourhood $N_L = N_G(s_1)\setminus K_4 =N_G(s_2) \setminus K_4$. If
\begin{equation} \label{eq::tetrahedron_neighbourhood_assignment}
\big( N_G(s_1)\setminus K_4 \big)
=
\big( N_G(s_2)\setminus K_4 \big)
\subset
\big( N_G(u)\setminus K_4 \big)
\subseteq
\big( N_G(v)\setminus K_4 \big),
\end{equation}
and if every vertex in the common neighbourhood of each leaf has no neighbours outside the tetrahedron and that neighbourhood itself,
\begin{equation} \label{eq::tetrahedron_leaf_class}
N_G(w) \subseteq K_4 \cup N_L \qquad \forall w\in N_L,
\end{equation}
then $u$ is designated as the internal vertex, $v$ as the external vertex, and $s_1,s_2$ as the leaf vertices of the tetrahedron.

Under Equations~\eqref{eq::tetrahedron_neighbourhood_assignment} and~\eqref{eq::tetrahedron_leaf_class}, the vertices $s_1$ and $s_2$ correspond to leaves adjacent to a common internal vertex $u$, whilst $v$ acts as the connecting vertex between $u$ and the remainder of the graph. The vertices with identical minimal neighbourhood structure correspond to leaves adjacent to a common internal vertex $u$, whilst the vertex $v$, having maximal neighbourhood containment, acts as the connecting vertex between $u$ and the remainder of the graph.

For $n=4$, the vertex $u$ is a binary branching vertex contained in a unique $4$-clique. In particular, when $n=5$, the branching vertex $u$ has three leaf children $s_1,s_2,s_3$, producing $\displaystyle {3 \choose 2} =3$ distinct $4$-cliques
\[
\{s_1,s_2,u,v\},\quad
\{s_1,s_3,u,v\},\quad
\{s_2,s_3,u,v\}.
\]

More generally, suppose $u$ has $k$ leaf children $s_1,\dots,s_k$, and that these vertices together with $v$ induce a $(k+2)$-clique. Then every pair of leaves $(s_i,s_j)$ determines a distinct $4$-clique $\{s_i,s_j,u,v\}$ where the total number of such $4$-cliques generated by this branching configuration is $ \displaystyle {k \choose 2}$. Suppose that $s_1,s_2$ are chosen without loss of generality, then if the $4$-clique satisfies Equation~\eqref{eq::tetrahedron_neighbourhood_assignment} and Equation~\eqref{eq::tetrahedron_leaf_class} can deterministically assign values to the edges using  
\begin{equation} \label{eq:tetrahedron_fix}
x_{s_1u}=x_{us_1} = x_{s_2u}=x_{us_2} = x_{uv}=x_{vu} = 1. 
\end{equation}

\begin{lemma} \label{lemma::tetrahedron_subset_neighbourhood}
  Let $s_1,s_2,u,v$ form a $4$-clique in square $G$. If $N_G(s_1)\setminus\{s_1,s_2,u,v\}=N_G(s_2)\setminus\{s_1,s_2,u,v\}$, then
  \[
  N_G(u)\setminus\{s_1, s_2\}=N_G(v)\setminus\{s_1, s_2\}.
  \]
  That is, $N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure in $G$.
\end{lemma}
\begin{proof}
  Let $x \in N_G(v)\setminus\{s_1,s_2\}$ be a vertex adjacent to $v$ in $G$ such that there exists a path of length at most two between $x$ and $v$ in $H$. If $x$ is adjacent to either $s_1$ or $s_2$, then by the assumption
  \[
  N_G(s_1)\setminus\{s_1,s_2,u,v\}
  =
  N_G(s_2)\setminus\{s_1,s_2,u,v\},
  \]
  where the vertices $s_1$ and $s_2$ contribute identically outside the clique. Hence any external connection from the clique through $v$ must also be realised through the same local structure containing $u$. Consequently $x$ is also at distance at most two from $u$ in $H$, and therefore
  \[
  x\in N_G(u)\setminus\{s_1,s_2\},
  \]
  which implies that $N_G(v)\setminus\{s_1,s_2\} \subseteq N_G(u)\setminus\{s_1,s_2\}$. By symmetry, interchanging the roles of $u$ and $v$ gives
  \[
  N_G(u)\setminus\{s_1,s_2\} \subseteq N_G(v)\setminus\{s_1,s_2\} 
  \]
  and hence  $N_G(u)\setminus\{s_1,s_2\} = N_G(v)\setminus\{s_1,s_2\}$. Therefore the vertices adjacent to $u$ and $v$ outside of $\{s_1,s_2\}$ have identical adjacency structure in $G$, and so $N_G(u)\cup\{u\}$ and $N_G(v)\cup\{v\}$ induce the same clique structure in $G$.
\end{proof}

\begin{proposition}[Tetrahedron Decomposition Interchangeability] \label{prop::textrahedron_decomposition_equality}
  Let $G$ be a graph that admits a graph root, and let $s_1,s_2,u,v$ form a $K_4$ in square $G$. If Equation~\eqref{eq::tetrahedron_neighbourhood_assignment} and Equation~\eqref{eq::tetrahedron_leaf_class} are satisfied and
  \[
  N_G(u)\setminus K_4=N_G(v)\setminus K_4,
  \]
  then $G$ admits a tree root $T$ such that $s_1u,s_2u,uv \in E_T$. 
\end{proposition}
\begin{proof}
  Since $s_1,s_2,u,v$ form a $K_4$ in $G$ and
  \[
  N_G(u)\setminus K_4=N_G(v)\setminus K_4,
  \]
  the vertices $u$ and $v$ have identical neighbourhoods outside $K_4$. Moreover, since $u$ and $v$ are both adjacent to every vertex of $K_4$, it follows that
  \[
  N_G(u)=N_G(v).
  \]
  Thus, exchanging $u$ and $v$ is an automorphism of $G$. Hence, if the root of $G$ is a tree root $T$ and has $v$ as the internal vertex of the tetrahedral configuration, applying this automorphism allows the roles of $u$ and $v$ to be exchanged without changing the square graph. Therefore, without loss of generality may assume that $u$ is the internal vertex of the tetrahedral configuration in the root $G$ and such that $s_1u,s_2u,uv \in E_T$
\end{proof}

\begin{proposition} \label{prop::tetrahedron_fixing}
  Let $H$ be a tree and let $u$ be an internal vertex with leaf children $s_1,\dots,s_k$ and parent $v$. Fixing any two leaves together with their internal vertex and parent uniquely determines the remaining leaf children of $u$ in the linear program.
\end{proposition}
\begin{proof}
  Since any tree may be rooted at an arbitrary vertex, assume without loss of generality that $H$ is rooted. Let $s_1,\dots,s_k$ be leaf children adjacent to an internal vertex $u$, and let $v$ be the parent of $u$. Consider any pair of leaves $s_i,s_j$, and suppose the internal branching vertex $u$ together with its parent $v$ have been fixed in the linear program. Now consider any remaining leaf $s_\ell$ adjacent to $u$. Since $s_\ell$ is a leaf in the root, its only neighbour in the tree is $u$. Furthermore, all leaf children of $u$ have identical neighbourhood structure in the square graph outside of $\{u,v,s_i,s_j\}$.

  Now consider any remaining leaf $s_\ell$ adjacent to $u$. Since every leaf adjacent to $u$ lies at distance two from $s_\ell$ through $u$, and $v$ lies at distance two through the path $s_\ell-u-v$, the neighbourhood of $s_\ell$ in the square graph is
  \[
  N_G(s_\ell) = \{u,v,s_1,\dots,s_{\ell-1},s_{\ell+1},\dots,s_k\}.
  \]

  Suppose for contradiction that $s_\ell$ is assigned to a vertex $w$ with $w\neq u$. Since $s_\ell$ is a leaf in the rooted tree, its neighbourhood in the square graph is determined by the vertices at distance at most two from $w$. Since $w\neq u$, the set of vertices at distance at most two from $s_\ell$ differs from that induced through $u$, giving
  \[
  N_G(s_\ell)
  \neq
  \{u,v,s_1,\dots,s_{\ell-1},s_{\ell+1},\dots,s_k\}.
  \]

  Hence either additional adjacencies are introduced into the square graph or existing adjacencies are removed and the neighbourhood of $s_\ell$ in the resulting square graph differs from that of the original graph, contradicting feasibility of the linear program. Thus the only feasible assignment is for $s_\ell$ to be adjacent to the already identified internal vertex $u$, implying that fixing any two leaves together with their internal vertex and parent uniquely determines all remaining leaf children adjacent to $u$.
\end{proof}

\subsection{Unicliqual Vertex Linear Program} \label{sec::unicliqual_vertex_linear_program}

Using Lemma~\eqref{lemma::tree_pendant_pendant_fork}, the formulations developed in Subsection~\ref{subsec::single_triangular_decomposition}--\ref{subsec::tetrahedron_decomposition} can be combined into one succinct linear program by first identifying a leaf-steam structure involving three vertices $s, u, v$. In particular, if a graph root exists, the structural properties of the graph guarantee that there is a root containing the identified edges. We may therefore safely fix the corresponding $x$-variables to $1$, thereby restricting the formulation to solutions containing these edges without excluding any graph root.

If $s$ is a leaf, $u$ is its corresponding stem, and there exists $v\in N_G[u]\setminus\{u\}$ such that
\[
N_G[s]\subset N_G[u]\subseteq N_G[v],
\]
with all vertices in $N_G[s]\setminus\{u,v\}$ having the same closed neighbourhood as $s$, then the corresponding root edges can be fixed by imposing
\begin{equation} \label{eq::unicliqual_vertex_fix}
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
\end{equation}

\begin{proposition}[Unicliqual Vertex Interchangeability] \label{prop::unicliqual_decomposition_equality}
  Let $s,u,v$ form a $K_3$ in square $G$, where $s$ is unicliqual. If $N_G[u]=N_G[v]$ then $u$ and $v$ may be chosen arbitrarily as the corresponding vertex in the tree root $T$ without loss of generality.
\end{proposition}
\begin{proof} 
  Since $s$ is unicliqual, the maximal clique containing $s$ is uniquely determined. If this clique satisfies the conditions of Proposition~\ref{prop::single_triangular_decomposition_equality}, then that proposition permits either $u$ or $v$ to be chosen as the corresponding vertex, since $N_G[u]=N_G[v]$ gives $\deg_G(u)=\deg_G(v)$. If the configuration instead satisfies the conditions of Proposition~\ref{prop::textrahedron_decomposition_equality}, then the equality $N_G[u]=N_G[v]$ gives
  \[
  N_G(u)\setminus K_4=N_G(v)\setminus K_4,
  \]
  so either $u$ or $v$ may be chosen in the tetrahedron fixing. Hence, in either case, $u$ and $v$ are interchangeable, and either may be chosen without loss of generality.
\end{proof}

It remains to verify that whenever a root $H$ of $G$ exists, the addition of the constraint Equation~\eqref{eq::unicliqual_vertex_fix} preserves the feasibility of the linear program.

\begin{proposition}[Validity of Unicliqual Vertex Fix] \label{prop::unicliqual_vertex_fix_feasibility}
  If a root $H$ of $G$ exists, then $\max \mathcal{UVLP}(G)$ is feasible.
\end{proposition}
\begin{proof}
  Suppose that a root $H$ of $G$ exists. By Proposition~\ref{prop::linear_relaxation_feasibility}, $\mathcal{LP}(G)$ is feasible. It remains to show that there exists a root whose adjacency matrix satisfies Equation~\eqref{eq::unicliqual_vertex_fix}.

  Let $s,u,v$ be the vertices identified by the unicliqual vertex fixing. By Proposition~\ref{prop::unicliqual_decomposition_equality}, if $s,u,v$ form the corresponding $K_3$ and $N_G[u]=N_G[v]$, then $u$ and $v$ may be chosen arbitrarily as the corresponding vertex in the tree root without loss of generality. Thus, if the given root $H$ does not satisfy Equation~\eqref{eq::unicliqual_vertex_fix}, the roles of $u$ and $v$ may be interchanged to obtain another root $H'$ of $G$ for which
  \[
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
  \]
  Hence the adjacency matrix of $H'$ is a feasible solution of $\mathcal{IP}(G)$ that satisfies Equation~\eqref{eq::unicliqual_vertex_fix}. Since every feasible solution of $\mathcal{IP}(G)$ is feasible for $\mathcal{LP}(G)$, this solution is also feasible for $\max \mathcal{UVLP}(G)$. Therefore, $\max \mathcal{UVLP}(G)$ is feasible whenever a root $H$ of $G$ exists.
\end{proof}

We first show that the unicliqual vertex linear program will always fix a sparse pendant or pendant fork. 

\begin{lemma} \label{lemma::unicliqual_vertex_fix}
  Let $G$ and $T$ be graphs such that $G = T^2$ and $T$ is a tree. $\max\mathcal{UVLP}(G)$ always fixes a sparse pendant or pendant fork subgraph of $T$.
\end{lemma}
\begin{proof}
  By Lemma~\ref{lemma::tree_pendant_pendant_fork}, $T$ either contains a sparse pendant or a pendant fork subgraph. Let $s$ be a leaf of this subgraph and let $u$ be its stem. Since $s$ is a leaf of $T$, every vertex of  $N_G[s]$ is either $s$, $u$, the neighbours of $u$ in $T$, or a vertex at distance two from $s$ through $u$. Hence
  \[
  N_G[s] \subset N_G[u].
  \]
  Moreover, if $v$ is the neighbour of $u$ in $T$ that connects the sparse pendant or pendant fork to the remainder of $T$, then every vertex adjacent to $u$ in $T$ is also within distance two of $v$ in $T$.
  This gives 
  \[
  N_G[u] \subseteq N_G[v].
  \]
  In the pendant fork case, the other leaf $s'$ of the fork satisfies $N_G[s']=N_G[s]$, since both $s$ and $s'$ are leaves adjacent to the same vertex $u$. Thus the vertices in $N_G[s]\setminus\{u,v\}$ have the same closed neighbourhood as $s$.

  Therefore the vertices $s,u,v$ satisfy the conditions for  Equation~\eqref{eq::unicliqual_vertex_fix}, which imposes
  \[
  x_{su}=x_{us}=x_{uv}=x_{vu}=1.
  \]
\end{proof}


\begin{lemma}[Propagation] \label{lemma::propagation}
  Let $G$ and $T$ be graphs such that $G=T^2$ and $T$ is a tree. Let $s,u,v \in V_T$ such that $s$ is a leaf, $u$ is its stem, and $v$ is the external vertex. Let $a_1,a_2,\dots a_r$ be the set of neighbours of $v$ in $T$. In every feasible solution $x$ of $\mathcal{UVLP}$, 
  \[
  x_{va_i}=x_{a_iv}=1 \text{ for every } i \in \{1,\ldots,r\}. 
  \]
\end{lemma}
\begin{proof}
  Since $s$ is a leaf and $u$ is a stem of $T$, the path $s-u-v$ is contained in $T$. Let $A=\{a_1,\ldots,a_r\}$ denote the vertices adjacent to $v$ in $T$ other than $u$, and let $B=\{b_1,\ldots,b_s\}$ denote the vertices adjacent to a vertex in $A$ other than $v$. For each $a_i\in A$, the vertices $s,u,v,a_i$ form a $P_4$ in $T$. Since $x_{su}=1$ and there exists no cycles in $T$, Lemma~\eqref{lemma::0_path_propagation} gives
  \[
  x_{ua_i}=0 \quad \text{ for all } a_i \in A.
  \]
  Using Lemma~\eqref{lemma::1_path_propagation} 
  This process can be repeated for each $a_i\in A$, giving $x_{va_i}=1$ for all $i\in\{1,\ldots,r\}$. Thus the edges $uv, va_1,\ldots,va_r$ are identified in the root, with $u$ acting as a leaf, $v$ as its stem, and each $a_i$ as the next vertex along the tree.
\end{proof}

\begin{theorem} \label{theorem::unicliqual_vertex_induction_proof}
  Let $G$ and $T$ be graphs such that $G=T^2$ and $T$ is a tree. Then $\mathcal{UVLP}(G)$ returns an integral solution.
\end{theorem}
\begin{proof}
  Suppose that $G=T^2$ for some tree $T$. We consider two cases. 
  
  First, suppose that $T$ is a star. Then $G$ is a complete graph and the maximum objective therefore admits an integral solution that is the complete graph itself. Unicliqual fixing is not applied and unnecessary. 

  Now suppose that $T$ is not a star. By Lemma~\ref{lemma::unicliqual_vertex_fix}, the unicliqual vertex fixing identifies either a sparse pendant or a pendant fork in $T$.
  
  In the sparse pendant case, the fixing identifies a leaf $s$, its stem $u$, and the external vertex $v$. Equation~\eqref{eq::unicliqual_vertex_fix} imposes 
  \[ 
  x_{su}=x_{us}=x_{uv}=x_{vu}=1 
  \] 
  in $\mathcal{UVLP}(G)$. 
  
  In the pendant fork case, the fixing identifies leaves $s_1,s_2$ with common parent $u$, together with the external vertex $v$ adjacent to $u$. The remaining leaves of $u$ need not be considered individually, as they form a clique with the other leaves and $u$ by the maximum objective function. Selecting one of the $s_1$ leaves as $s$ likewise imposes 
  \[ 
  x_{s_1u}=x_{us_1}=x_{uv}=x_{vu}=1 
  \] 
  in $\mathcal{UVLP}(G)$. 
  
  Let $N_T(v)\setminus\{u\} = \{a_1,\ldots,a_r\}$. Hence the hypotheses of Lemma~\ref{lemma::propagation} are satisfied and the path propagation lemma implies
  \[
  x_{va_i}=x_{a_iv}=1 \text{ for every } i \in \{1,\ldots,r\},
  \]
  for every $a_i \in N_T(v)\setminus\{u\}$. 

  We prove by induction on the number of edges of the reconstructed subtree that repeated application of the path propagation lemma assigns value $1$ to every edge of $T$. 
    
  For the base case, the reconstructed subtree consists of the edges initially identified by the unicliqual vertex fixing and the first application of path propagation. These edges have integral assignments by Equation~\eqref{eq::unicliqual_vertex_fix} and Lemma~\ref{lemma::propagation}. 
  
  For the inductive step, suppose that a connected subtree $T'$ of $T$ has been reconstructed and that every edge of $T'$ has been assigned its integral value. If $T'\neq T$, since $T$ is connected, there exists an edge 
  \[ 
  ab \in E_T \setminus E_{T'} 
  \] 
  with $a \in V_{T'}$ and $b \notin V_{T'}$. Since $T'$ contains at least one path of length two, we may arbitrarily label three consecutive vertices of such a path as $s,u,v$. The vertex $a$ is then the already reconstructed vertex adjacent to the unlabelled vertex $b$. Relabelling the vertices in this local configuration in this way, the hypotheses of Lemma~\ref{lemma::propagation} are satisfied. Hence path propagation fixes 
  \[ 
  x_{ab}=x_{ba}=1. 
  \] 
  Thus $T'$ can be extended to the larger connected subtree $T''=T'+ab$, whose edges all have integral assignments. 
  
  Thus, by the inductive argument, whenever the reconstructed subtree is a proper subtree of $T$, it can be extended by at least one additional edge of $T$, with all edges of the resulting subtree assigned integral values. Since $T$ is finite, the induction eventually reconstructs all of $T$ and assigns an integral value to every edge. Hence the resulting solution is integral and corresponds to the tree root $T$ of $G$. Therefore, $\max\mathcal{UVLP}(G)$ returns an integral solution.
\end{proof}

\begin{definition}[Cycle Amalgamation] \label{def::cycle_amalgamation}
  A \textbf{cycle amalgamation} or an \textbf{amalgamation of cycles} is a graph whereby every vertex is cyclic, but there exists more than one cycle. An amalagmation on $n$ vertices can be denoted symbolically by
  \[
  C_aC_{b-i}C_{c-j}C_{d-k} \ldots
  \]
  where $n = a + (b - i) + (c - j) + (d - k) + \ldots $ and $C_aC_bC_cC_d \ldots$ denote that there are subcycles of length $a,b,c,d \ldots$ and that $C_b$ shares $i$ vertices with $C_a$ and $C_c$ shares $j$ vertices with $C_aC_{b-i}$ and so on. 
\end{definition}

The degree pattern of the cyclic core can distinguish several possible cycle structures. For example, suppose $n = |V_C| = 6$. Then the following structures give the indicated degree sequences.:
\begin{enumerate}[label=\roman*)]
  \item If the root subgraph is the amalgamation of two $5$-cycles sharing four vertices $C_5C_{5-4}$ or is the amalgamation of three $4$-cycles sharing three vertices $C_4C_{4-3}C_{4-3}$, then the square subgraph has
  \[
  \deg_{G[C]}(v)=5 \text{ for every } v \in V_C,
  \]
  \item If the subgraph of the root is $C_6$, then the subgraph of the square $G[C]$ has
  \[
  \deg_{G[C]}(v)=4 \text{ for every } v \in V_C,
  \]
  \item If the root subgraph is the amalgamation of two $4$-cycles sharing an edge $uv$ with two vertices $C_4C_{4-2}$, then the shared vertices $u$ and $v$ have degree $5$, while the remaining four vertices have degree $4$. Thus
  \[
  \deg_{G[C]}(u)=\deg_{G[C]}(v)=5, \deg_{G[C]}(w)=4 \text{ for } w \in V_C \setminus\{u,v\},
  \]
\end{enumerate}

\begin{algorithm}\caption{Cyclic-Core Root Reconstruction} \label{algo::cyclic_core_reconstruction_1}
  \begin{algorithmic}[1]
    \State \textbf{Input:} A graph $G$
    \State \textbf{Output:} Set of edges $F$ to fix

    \vspace{0.5em}

    \Statex \textbf{Step 1: Iterative unicliqual vertex removal}
    \State Set $V_0 = V(G)$, $i=1$, and initialize $\mathcal{L}=\emptyset$
    \While{$V_{i-1}\neq\emptyset$}
      \State Let $G_i=G[V_{i-1}]$. 
      \State Identify when $U_i=\{v\in V_{i-1}: \text{v is unicliqual} \}$ or $U_i=\{v\in V_{i-1}: G_i[N_{G_i}(v)]\text{ is complete}\}$.
      \If{there are no unicliqual vertices, $U_i=\emptyset$}
          \State Set $C=V_{i-1}$ and $\mathsf{AUQ}=$ FALSE and \textbf{break}
      \ElsIf{every vertex is unicliqual, $U_i=V_{i-1}$}
          \State Set $C=V_{i-1}$ and $\mathsf{AUQ}=$ TRUE \textbf{break}
      \Else
          \State Add $U_i$ to the ordered collection $\mathcal{L}$
          \State Set $V_i=V_{i-1}\setminus U_i$
          \State For each $v\in U_i$, record $N_i(v)=N_G(v) \cap V_i$
          \State Set $i\leftarrow i+1$
      \EndIf
    \EndWhile
    \State $C$ is the \emph{cyclic core} and $\mathcal{L}$ contains the successive unicliqual \emph{layers} $U_i$.

    \vspace{0.5em}

    \Statex \textbf{Step 2: Construct the core graph}

    \State Let $H=G[C]$.
    \State Compute the degree $d_C(v)=d_H(v)$ for every $v\in C$.
    \State Let $U_k$ be the last removed unicliqual layer, if one exists.
    \If{$U_k$ exists}
        \State For each $u\in U_k$, compute the recorded core neighbourhood $N_k(u)\cap C$.
        \If{there exists $u\in U_k$ such that $|N_k(u)\cap C|\geq4$}
            \State \textbf{Cycle amalgamation with $U_k$}
        \Else
            \State \textbf{Cyclic core with $U_k$}
        \EndIf
    \Else
        \If{all vertices of $C$ have the same degree $d\leq4$}
            \State \textbf{Uniform cyclic core without $U_k$}
        \Else
            \State \textbf{Cycle amalgamation without $U_k$}
        \EndIf
    \EndIf
    \algstore{P1}
  \end{algorithmic}
\end{algorithm}

\begin{algorithm} \ContinuedFloat \caption{Cyclic-Core Root Reconstruction (continued)} \label{algo::cyclic_core_reconstruction_2}
  \begin{algorithmic}[1]

  \algrestore{P1}

  \Statex \textbf{Step 3: Cycle amalgamation with $U_k$}
  \If{$U_k$ exists and $\displaystyle\max_{u\in U_k}|N_k(u)\cap C|\geq4$}
      \State Partition $U_k$ by $u\sim v\iff N_k(u)\cap C=N_k(v)\cap C$.
      \State Initialize $B_C$ with $E(B_C)=\emptyset$.
      \State Process the groups in non-increasing order of $|N_k(u)\cap C|$, ties by vertex label.
      \For{each group $S$}
          \State Let $N_S=N_k(u)\cap C$ for any $u\in S$ and decompose $S$ into $G$-components $S_1,\ldots,S_m$.
          \State Process $S_1,\ldots,S_m$ greedily in non-increasing order of the degree of their admissible attachment vertices in $H$.
          \State For $S_1$, choose an attachment $c_1\in N_S$ by the degree rule, choosing a maximum-degree attachment when $|N_S|\geq4$ and the appropriate minimum-degree attachment when $|N_S|\leq3$, and choose the corresponding cyclic orientation arbitrarily.
          \State Fix $uc_1\in E(B_C)$ for every $u\in S_1$ and extend $B_C$ according to this orientation.
          \For{$j=2,\ldots,m$}
              \State Choose $c_j\in N_S$ greedily so that $c_j$ and the induced cyclic orientation are consistent with the orientation already fixed and $B_C^2\subseteq H$.
              \State Fix $uc_j\in E(B_C)$ for every $u\in S_j$ and extend the existing orientation.
          \EndFor
      \EndFor
      \State Add $E(B_C)$ to $F$.
  \EndIf

  \Statex \textbf{Step 4: Cyclic core with $U_k$}
  \If{$U_k$ exists and $\displaystyle\max_{u\in U_k}|N_k(u)\cap C|\leq3$}
      \State Partition $U_k$ by $u\sim v\iff N_k(u)\cap C=N_k(v)\cap C$ and initialize $B_C$ with $E(B_C)=\emptyset$.
      \For{each group $S$}
          \State Let $N_S=N_k(u)\cap C$ for any $u\in S$ and decompose $S$ into $G$-components $S_1,\ldots,S_m$.
          \If{$|N_S|=3$}
              \State Assign distinct attachments $c_1,\ldots,c_m\in N_S$ to $S_1,\ldots,S_m$.
              \For{$j=1,\ldots,m$}
                  \State Fix $uc_j\in E(B_C)$ for every $u\in S_j$ and add the corresponding cycle edges on $N_S$ through $c_j$ to $B_C$.
              \EndFor
          \EndIf
      \EndFor
      \State Add $E(B_C)$ to $F$.
  \EndIf

  \algstore{P2}
  \end{algorithmic}
\end{algorithm}
\begin{algorithm} \ContinuedFloat \caption{Cyclic-Core Root Reconstruction (continued)} \label{algo::cyclic_core_reconstruction_3}
  \begin{algorithmic}[1]

  \algrestore{P2}

  \Statex \textbf{Step 5: Uniform cyclic core without $U_k$}
  \If{$U_k$ does not exist and all vertices of $C$ have the same degree $d\leq4$}
      \State Select arbitrary $v\in C$ and initialize $B_C$ with $E(B_C)=\emptyset$.
      \If{$d\leq3$}
          \State Select arbitrary $c_1\in N_H(v)$, fix $vc_1\in E(B_C)$, and propagate the reconstruction subject to $B_C^2=H$.
      \ElsIf{$d=4$}
          \State Let $N_H(v)=\{c_1,c_2,c_3,c_4\}$ and arbitrarily fix $vc_1,vc_2\in E(B_C)$.
          \State Use $c_3,c_4$ and $B_C^2=H$ to determine the remaining root edges incident with $v$.
          \State Propagate the reconstruction through every unresolved vertex of $C$.
      \EndIf
      \State Add $E(B_C)$ to $F$.
  \EndIf

  \Statex \textbf{Step 6: Cycle amalgamation without $U_k$}
  \If{$U_k$ does not exist and the vertices of $C$ do not all have the same degree $d\leq4$}
      \State Let $v,u\in C$ be adjacent vertices with $d_H(v)=d_H(u)=\Delta=\max_{x\in C}d_H(x)$.
      \State Fix $vu\in E(B_C)$.
      \State Choose distinct $a,b\in N_H(v)\setminus\{u\}$ and distinct $c,d\in N_H(u)\setminus\{v\}$ such that $\{a,b,c,d\}$ are pairwise distinct and $(B_C\cup\{va,vb,uc,ud\})^2\subseteq H$.
      \State Fix $va,vb,uc,ud\in E(B_C)$ and add these forced edges to $F$.
  \EndIf

  \Statex \textbf{Step 7: Output}
  \State \textbf{Return} $F$.

  \end{algorithmic}
\end{algorithm}
